\section{Population Model and Induced Network}
\label{sec:model}

\subsection{Assignment state (relaxed robot counts $z$)}
A team of $N=\sum_\tau n_\tau$ robots is partitioned into $T$ types
$\tau\in\{1,\dots,T\}$ (heterogeneous in force capacity and communication range),
with $n_\tau$ robots of type $\tau$. There are $M$ loads with $H$ contact slots
each, a set $\mathcal R$ of candidate relay sites, and the common action set
\begin{equation}
\mathcal A=\underbrace{\{(k,h)\}_{k\le M,\,h\le H}}_{\text{load--slot}}
\;\cup\;\mathcal R\;\cup\;\{\mathrm{o}\},
\label{eq:actions}
\end{equation}
with $\mathrm o$ idle. We write $a\in\mathcal A$ for a generic action, $(k,h)$ for a
load--slot, and $r\in\mathcal R$ for a relay site.

The \emph{assignment state} is the relaxed-count matrix
$z=(z_{\tau a})\in\R_{\ge0}^{T\times|\mathcal A|}$: $z_{\tau a}$ is a continuous
relaxation of the number of type-$\tau$ robots assigned to action $a$---neither a
probability nor a physical force. Each type spends its budget, and each slot or
relay site holds at most one robot,
\begin{equation}
\begin{aligned}
&\textstyle\sum_{a\in\mathcal A} z_{\tau a}=n_\tau\ \ \forall\tau,
\qquad z_{\tau a}\ge0,\\[1mm]
&\textstyle\sum_\tau z_{\tau,kh}\le1\ \ \forall(k,h),
\qquad \sum_\tau z_{\tau,r}\le1\ \ \forall r.
\end{aligned}
\label{eq:conservation}
\end{equation}
The feasible set
$\mathcal X=\{z\in\R_{\ge0}^{T\times|\mathcal A|}:\eqref{eq:conservation}\}$ is a
compact convex polytope. A finite team instantiates $z$ by an integer assignment
(one action per robot); integerization is treated in Remark~\ref{rem:integer}. The
state $z$ is distinct from the physical poses $q\in(\mathrm{SE}(2))^N$, which enter
only the closed-loop simulation (Section~\ref{sec:exp}). Multiplying a count
$z_{\tau kh}$ by a per-robot force yields a total force, so every aggregate below is
dimensionally a force or torque.

\begin{assumption}[Standing hypotheses]\label{as:standing}
Throughout: \emph{(i)} $\mathcal X$ is compact and convex; \emph{(ii)} the
wrench-feasible set $\mathcal X_{\mathrm f}$ of Lemma~\ref{prop:membership} is a
polytope;
\emph{(iii)} the candidate-site Laplacian $\Lap(z)$ of \eqref{eq:laplacian} is affine
in $z$ with each elementary $L_e\succeq0$ and each coefficient $\ge0$; \emph{(iv)}
the communication graph is undirected and weighted, so every $\Lap(z)$ is a symmetric
graph Laplacian; \emph{(v)} $\mathcal X_{\mathrm f}\neq\varnothing$; and \emph{(vi)}
wherever the gradient $\nabla\lamtwo$ is used, $\lamtwo(\Lap(z))$ is a \emph{simple}
eigenvalue (a weak background regulariser $\delta_0(I-\tfrac1N\mathbf 1\mathbf 1^{\top})$
enforces this in practice); at multiplicity the Fiedler gradient is replaced by the
PSD matrix dual $Z\succeq0$ of the LMI~\eqref{eq:lmi}.
\end{assumption}

\subsection{Exact wrench feasibility}
A type-$\tau$ robot with heading $\phi$ exerts contact forces in the heading-aligned
ellipse $U_\tau(\phi)$ with semi-axes $F_\tau$ (along $\phi$) and $\kappa F_\tau$
($\kappa\le1$, the nonholonomic penalty). The strategic layer does not track headings,
so we certify against the heading-independent \emph{inner zonotope}
$U_\tau^{\mathrm{in}}=\{G_\tau u:\|u\|_\infty\le1\}$ (generator matrix $G_\tau$)
inscribed in the disk of radius $\kappa F_\tau$. Since
$\mathrm{disk}(\kappa F_\tau)\subseteq U_\tau(\phi)$ for every $\phi$, we have
$U_\tau^{\mathrm{in}}\subseteq U_\tau(\phi)$, so feasibility tested against
$U_\tau^{\mathrm{in}}$ is \emph{conservative} and creates no false positive.

Let $A_{kh}\in\R^{3\times2}$ be the contact-to-wrench map at slot $(k,h)$ and
$\wdem_k\in\R^3$ the demanded wrench of load $k$. The reachable population wrench set is
the scaled Minkowski sum
$\Wmap(C_k)=\bigoplus_{\tau,h}z_{\tau kh}A_{kh}U_\tau^{\mathrm{in}}$.

\begin{lemma}[Exact relaxed wrench feasibility]\label{prop:membership}
$\wdem_k\in\Wmap(C_k)$ iff there exist generator coefficients $\xi_{\tau kh}$ with
\begin{equation}
\sum_{\tau,h}A_{kh}G_\tau\,\xi_{\tau kh}=\wdem_k,
\qquad
-z_{\tau kh}\mathbf 1\le\xi_{\tau kh}\le z_{\tau kh}\mathbf 1 .
\label{eq:tensor}
\end{equation}
Wrench feasibility of load $k$ is thus a linear feasibility problem in $(z,\xi)$, and
its projection onto $z$, $X_{\mathrm f}=\{z\in\mathcal X:H_w z\ge d\}$, is a polytope.
\end{lemma}
\begin{proof}
$z_{\tau kh}A_{kh}U_\tau^{\mathrm{in}}=\{A_{kh}G_\tau\xi:\|\xi\|_\infty\le z_{\tau kh}\}$
by linear image and box scaling; the Minkowski sum over $(\tau,h)$ superposes
independent generators, giving \eqref{eq:tensor}. The lifted set is linear in
$(z,\xi)$, and eliminating $\xi$ (projection of a polyhedron) yields the polytope
$X_{\mathrm f}$.
\end{proof}
By $U_\tau^{\mathrm{in}}\subseteq U_\tau(\phi)$ a certified wrench is realizable by the
physical elliptical forces. The certificate is quasi-static and assumes bilateral,
independent contact forces; it does not model frictional grasp or unilateral pushing.

\subsection{Decision-induced candidate-site Laplacian $\Lap(z)$}
We separate two graphs throughout, both on the \emph{same} node set: each of the $N$
robots is a candidate-site node ($V=N$), and $\Lap(z)$ and $\Lgeo(q)$ index it
identically, so the mismatch $\|\Lgeo(q)-\Lap(z)\|_2$ of Lemma~\ref{lem:bridge} is
well defined. The \emph{decision-induced candidate-site Laplacian} $\Lap(z)$ depends
only on the assignment $z$. A fixed base graph $L_0\succeq0$ carries the always-on
intra-cluster links;
occupying relay site $r$ with a type-$\tau$ robot switches on the link Laplacian
$L_r\succeq0$ with effectiveness $\gamma_{\tau r}\ge0$ (long-range types reach across
the gap, short-range types barely), so
\begin{equation}
\Lap(z)=L_0+\sum_{\tau,r}\gamma_{\tau r}\,z_{\tau r}\,L_r\ \in\R^{V\times V},
\label{eq:laplacian}
\end{equation}
affine in $z$ (Assumption~\ref{as:standing}(iii)). This is a convex
\emph{surrogate}: the actual, \emph{geometric} Laplacian $\Lgeo(q)$, whose weights
depend on inter-robot distances through the physical poses $q$, is nonconvex in
$q$ and is \emph{not} covered by the convex theory of
Sections~\ref{sec:wise}--\ref{sec:algorithm}; it is exercised only in the
closed-loop simulation, where we report the empirical agreement between
$\Lgeo(q)$ and its surrogate $\Lap(z)$. All theorems below use $\Lap(z)$ only.

Let $Q\in\R^{N\times(N-1)}$ be an orthonormal basis of $\mathbf 1^{\perp}$; the
information constraint is the LMI
\begin{equation}
Q^{\top}\Lap(z)\,Q\succeq\sigmastar I
\;\Longleftrightarrow\;
\lamtwo(\Lap(z))\ge\sigmastar,
\label{eq:lmi}
\end{equation}
affine in $z$, with PSD matrix dual $Z\succeq0$ (a scalar connectivity price
$\pi$ when $\lamtwo$ is simple). The requirement $\sigmastar=\sigmadyn+\delta$ pairs the
derived stability boundary $\sigmadyn$ (Theorem~\ref{thm:stability}) with the robust
margin $\delta$ of Lemma~\ref{lem:bridge}.

\begin{lemma}[Convex information set]\label{lem:convex}
$z\mapsto\lamtwo(\Lap(z))$ is concave, so \eqref{eq:lmi} defines a convex set.
\end{lemma}
\begin{proof}
$\lamtwo(L)=\min_{\|v\|=1,\,v\perp\mathbf 1}v^{\top}Lv$ is a pointwise minimum of
functions linear in $L$~\cite{fiedler1973}, hence concave; composed with the affine
$\Lap(z)$ it stays concave, and its super-level set is
convex~\cite{ghosh2006growing}.
\end{proof}

\begin{lemma}[Geometric transfer under model mismatch]\label{lem:bridge}
Suppose robots track their intended candidate sites so that
$\|\Lgeo(q)-\Lap(z)\|_2\le\varepsilon_L$. Then Weyl's inequality gives
\begin{equation}
\lamtwo(\Lgeo(q))\;\ge\;\lamtwo(\Lap(z))-\varepsilon_L .
\label{eq:weyl}
\end{equation}
Consequently, with the design margin
$\delta=\varepsilon_L+\varepsilon_{\mathrm{est}}+\delta_{\mathrm{num}}$
($\varepsilon_L$: geometric/model mismatch; $\varepsilon_{\mathrm{est}}$:
distributed eigenvalue-estimation error; $\delta_{\mathrm{num}}$: numerical
margin), the WISE requirement $\lamtwo(\Lap(z))\ge\sigmastar:=\sigmadyn+\delta$
implies $\lamtwo(\Lgeo(q))\ge\sigmadyn$, the physical stability threshold.
\end{lemma}
\begin{proof}
Weyl's inequality for symmetric matrices gives
$|\lamtwo(\Lgeo)-\lamtwo(\Lap)|\le\|\Lgeo-\Lap\|_2\le\varepsilon_L$, which is
\eqref{eq:weyl}; subtracting $\varepsilon_{\mathrm{est}}+\delta_{\mathrm{num}}$
absorbs the estimator and numerical errors.
\end{proof}
The convex theory of Sections~\ref{sec:wise}--\ref{sec:algorithm} uses the affine
$\Lap(z)$; it transfers to the geometric $\Lgeo(q)$ \emph{only} through
\eqref{eq:weyl} under the margin $\delta$, with $\varepsilon_L$ measured
(Section~\ref{sec:exp}, bounded by $C_L\rho$) and $\varepsilon_{\mathrm{est}},
\delta_{\mathrm{num}}$ fixed constants (Section~\ref{sec:exp}).
