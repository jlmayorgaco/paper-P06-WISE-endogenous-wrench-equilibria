\section{Assignment, Wrench, and Induced Network Model}
\label{sec:model}

\subsection{Assignment state (relaxed per-robot occupancy $z$)}
$N$ robots of $T$ types ($\tau(i)$ the type of robot $i$), $M$ loads with $H$ contact slots
each, relay sites $\mathcal R$, action set
$\mathcal A=\{(k,h)\}\cup\mathcal R\cup\{\mathrm o\}$ ($\mathrm o$ idle). The \emph{assignment
state} is the relaxed occupancy $z=(z_{ia})\in[0,1]^{N\times|\mathcal A|}$: each robot spends a
unit budget, each slot or relay site holds at most one robot,
\begin{equation}
\textstyle\sum_{a} z_{ia}=1\ \forall i,\quad
\sum_i z_{i,kh}\le1\ \forall(k,h),\quad \sum_i z_{i,r}\le1\ \forall r.
\label{eq:conservation}
\end{equation}
Type aggregates $z_{\tau a}=\sum_{i\in\mathcal N_\tau}z_{ia}$ enter the productive and wrench
layers; the communication layer uses the per-robot relay occupancies $z_{ir}$. The feasible set
$\mathcal X$ of \eqref{eq:conservation} is a compact convex polytope in $\R^{n}$,
$n=N|\mathcal A|$; poses enter only the geometric instantiation.

\begin{assumption}[Standing hypotheses]\label{as:standing}
\emph{(i)} $\mathcal X$ is compact convex, $\mathcal X_{\mathrm f}\neq\varnothing$;
\emph{(ii)} $\Lap(z)$ is affine in $z$, each $L_{ir}\succeq0$ symmetric and fixed;
\emph{(iii)} $\nabla\lamtwo$ is used only where $\lamtwo$ is simple, else the eigenspace
derivative (Def.~\ref{def:gamma}); \emph{(iv)} the physical graph decomposes additively as
$\Lgeo(q,\hat z)=L_0^{\mathrm{geo}}(q)+\sum_{i,r}\hat z_{ir}L_{ir}^{\mathrm{geo}}(q)$ with
$L_0^{\mathrm{geo}}(q)\succeq L_0$ and $L_{ir}^{\mathrm{geo}}(q)\succeq L_{ir}$ at every
admissible pose, no link double-counted.
\end{assumption}

\subsection{Wrench feasibility}
We assume \emph{quasi-static planar transport with rigid (bilateral) robot--load attachment},
so a robot may push \emph{and} pull along the modeled directions; unilateral pushing, friction
cones and contact-mode transitions are outside this model~\cite{murray1994manipulation}. The
capacity weight $c_\tau$ in $Bz$ is a scheduling coefficient, distinct from the actuation data
$(F_\tau,G_\tau,A_{kh})$.

A type-$\tau$ robot with heading $\phi$ exerts contact forces in the heading-aligned ellipse
$U_\tau(\phi)$, semi-axes $F_\tau,\kappa F_\tau$ ($\kappa\le1$). Not tracking headings, we
certify against the heading-independent \emph{inner zonotope}
$U_\tau^{\mathrm{in}}=\{G_\tau u:\|u\|_\infty\le1\}$ inscribed in
$\mathrm{disk}(\kappa F_\tau)\subseteq U_\tau(\phi)$: conservative, no false positive. With
$A_{kh}$ the contact-to-wrench map and $\wdem_k$ the demand,
$\Wmap(C_k)=\bigoplus_{\tau,h}z_{\tau kh}A_{kh}U_\tau^{\mathrm{in}}$.

\begin{lemma}[Exact lifted representation of the inner-zonotope actuation model]
\label{prop:membership}
$\wdem_k\in\Wmap(C_k)$ iff there exist generator coefficients $\xi_{\tau kh}$ with
\begin{equation}
\sum_{\tau,h}A_{kh}G_\tau\,\xi_{\tau kh}=\wdem_k,
\qquad
-z_{\tau kh}\mathbf 1\le\xi_{\tau kh}\le z_{\tau kh}\mathbf 1 .
\label{eq:tensor}
\end{equation}
Wrench feasibility of load $k$ is thus a linear feasibility problem in $(z,\xi)$: the
\emph{exact} membership test used to independently re-evaluate every reported candidate. It
follows by superposition of the scaled generators $z_{\tau kh}A_{kh}U_\tau^{\mathrm{in}}$.
\end{lemma}
Write $\widehat X$ for the lifted set in $(z,\xi)$ and $\prz(z,\xi)=z$, so
$X_{\mathrm f}=\prz(\widehat X)$: an \emph{extended formulation} rather than an inequality
description in $z$. Sec.~\ref{sec:wise} shows the relevant tangent cone projects from it
\emph{exactly}, so both Stage~2 and $\Gamma_{\Emanifold}$ avoid enumerating the projected
wrench facets. A finite-direction support polytope $\{z:H_wz\ge d\}$ serves only as a fast \emph{outer}
screen: candidates it admits are certified by \eqref{eq:tensor}.

\subsection{Decision-induced robot-graph Laplacian $\Lap(z)$}
The information graph has a \emph{fixed} node set, one node per robot: actions choose
\emph{edges} at a node, not nodes. A fixed base graph $L_0\succeq0$ carries the always-on
short-range conductances; committing robot $i$ to relay $r$ \emph{gates on} its predefined
long-range links
$L_{ir}=\gamma_{\tau(i)r}\sum_{j}\underline w^{r}_{ij}(e_i{-}e_j)(e_i{-}e_j)^{\top}\succeq0$,
weighted by tube infima $\underline w^{r}_{ij}=\inf_{q\in\mathcal Q_{ir}}w_{ij}(q)$ fixed before
optimization, so each $L_{ir}$ is decision-independent and the SDP stays affine. For $L_0$ the
infimum is over each robot's admissible positions \emph{across all its actions}, so both $L_0$
and every $L_{ir}$ are lower-bound Laplacians and $\Lgeo(q)\succeq\Lap(\hat z)$
(Assumption~\ref{as:standing}(iv)). Thus
\begin{equation}
\Lap(z)=L_0+\sum_{i,r}\,z_{ir}\,L_{ir}\ \in\R^{N\times N}.
\label{eq:laplacian}
\end{equation}
With $Q\in\R^{N\times(N-1)}$ an orthonormal basis of $\mathbf 1^{\perp}$, the information
constraint is the LMI
\begin{equation}
Q^{\top}\Lap(z)\,Q\succeq\sigmastar I
\;\Longleftrightarrow\;
\lamtwo(\Lap(z))\ge\sigmastar,
\label{eq:lmi}
\end{equation}
affine in $z$, with PSD dual $Y\succeq0$ whose trace $\pilam=\operatorname{tr}Y$ is the
connectivity price; $\sigmadyn$ is \emph{derived} in
Prop.~\ref{thm:stability} and $\sigmastar>\sigmadyn$ is a design margin. As
$\lamtwo(\Lap(z))=\min_{\|v\|=1,v\perp\mathbf1}v^{\top}\Lap(z)v$ is a pointwise minimum of
affine functions~\cite{fiedler1973} it is \emph{concave}, so \eqref{eq:lmi} is
convex~\cite{ghosh2006growing}; we use this repeatedly.

\begin{lemma}[A priori geometric transfer by graph monotonicity]\label{lem:bridge}
Let $\hat z\in\mathcal X\cap\{0,1\}^{n}$ be an integer recovery and $q$ any admissible pose. By
Assumption~\ref{as:standing}(iv) and the lower-bound construction of $L_0,L_{ir}$,
\begin{equation}
\Lgeo(q)-\Lap(\hat z)=(L_0^{\mathrm{geo}}{-}L_0)+\textstyle\sum_{i,r}\hat z_{ir}
(L_{ir}^{\mathrm{geo}}{-}L_{ir})\succeq0,
\label{eq:mismatch}
\end{equation}
a sum of PSD terms, so on $\mathbf 1^{\perp}$ eigenvalue monotonicity gives
$\lamtwo(\Lgeo(q))\ge\lamtwo(\Lap(\hat z))$: a certificate
$\lamtwo(\Lap(\hat z))\ge\sigmastar>\sigmadyn$ yields $\lamtwo(\Lgeo(q))\ge\sigmadyn$
\emph{a priori} at every admissible pose.
\end{lemma}
Every candidate is \emph{independently re-evaluated} in double precision against equality
residuals, slacks, wrench feasibility and the spectral margin: no dependence on
solver-reported status, but not exact-arithmetic certification.
