\section{Population Model and Induced Network}
\label{sec:model}

\subsection{Assignment state (relaxed robot counts $z$)}
A team of $N=\sum_\tau n_\tau$ robots is partitioned into $T$ types
$\tau\in\{1,\dots,T\}$ (heterogeneous in force capacity and communication range),
with $n_\tau$ robots of type $\tau$. There are $M$ loads with $H$ contact slots
each, a set $\mathcal R$ of candidate relay sites, and the common action set
\begin{equation}
\mathcal A=\underbrace{\{(k,h)\}_{k\le M,\,h\le H}}_{\text{load--slot}}
\;\cup\;\mathcal R\;\cup\;\{\mathrm{o}\},
\label{eq:actions}
\end{equation}
with $\mathrm o$ idle. We write $a\in\mathcal A$ for a generic action, $(k,h)$ for a
load--slot, and $r\in\mathcal R$ for a relay site.

The \emph{assignment state} is the relaxed-count matrix
$z=(z_{\tau a})\in\R_{\ge0}^{T\times|\mathcal A|}$: $z_{\tau a}$ is a continuous
relaxation of the number of type-$\tau$ robots assigned to action $a$---neither a
probability nor a physical force. Each type spends its budget, and each slot or
relay site holds at most one robot,
\begin{equation}
\begin{aligned}
&\textstyle\sum_{a\in\mathcal A} z_{\tau a}=n_\tau\ \ \forall\tau,
\qquad z_{\tau a}\ge0,\\[1mm]
&\textstyle\sum_\tau z_{\tau,kh}\le1\ \ \forall(k,h),
\qquad \sum_\tau z_{\tau,r}\le1\ \ \forall r.
\end{aligned}
\label{eq:conservation}
\end{equation}
The feasible set
$\mathcal X=\{z\in\R_{\ge0}^{T\times|\mathcal A|}:\eqref{eq:conservation}\}$ is a
compact convex polytope. A finite team instantiates $z$ by an integer assignment
(one action per robot); integerization is treated in Remark~\ref{rem:integer}. The
state $z$ is distinct from the physical poses $q\in(\mathrm{SE}(2))^N$, which enter
only the closed-loop simulation (Section~\ref{sec:exp}). Multiplying a count
$z_{\tau kh}$ by a per-robot force yields a total force, so every aggregate below is
dimensionally a force or torque.

\begin{assumption}[Standing hypotheses]\label{as:standing}
Throughout: \emph{(i)} $\mathcal X$ is compact convex and $\mathcal X_{\mathrm f}\neq
\varnothing$; \emph{(ii)} $\Lap(z)$ of \eqref{eq:laplacian} is affine in $z$ with each
$L_r\succeq0$ and coefficient $\ge0$, undirected (symmetric); \emph{(iii)} the
single-Fiedler gradient $\nabla\lamtwo$ is used only where $\lamtwo$ is simple with gap
$\lambda_3-\lamtwo\ge g_{\min}>0$; at multiplicity it is replaced by the eigenspace
directional derivative (Prop.~\ref{prop:freeconn}), while the SDP~\eqref{eq:lmi} itself stays
exact and needs no spectral differentiability (no genericity assumed).
\end{assumption}

\subsection{Wrench feasibility}
A type-$\tau$ robot with heading $\phi$ exerts contact forces in the heading-aligned
ellipse $U_\tau(\phi)$ with semi-axes $F_\tau$, $\kappa F_\tau$ ($\kappa\le1$,
nonholonomic penalty). The strategic layer does not track headings, so we certify
against the heading-independent \emph{inner zonotope}
$U_\tau^{\mathrm{in}}=\{G_\tau u:\|u\|_\infty\le1\}$ (generators $G_\tau$) inscribed in
the disk of radius $\kappa F_\tau$. Since $\mathrm{disk}(\kappa F_\tau)\subseteq
U_\tau(\phi)$, $U_\tau^{\mathrm{in}}\subseteq U_\tau(\phi)$, so feasibility tested
against
$U_\tau^{\mathrm{in}}$ is \emph{conservative} and creates no false positive.

Let $A_{kh}\in\R^{3\times2}$ be the contact-to-wrench map at slot $(k,h)$ and
$\wdem_k\in\R^3$ the demanded wrench of load $k$. The reachable population wrench set is
the scaled Minkowski sum
$\Wmap(C_k)=\bigoplus_{\tau,h}z_{\tau kh}A_{kh}U_\tau^{\mathrm{in}}$.

\begin{lemma}[Exact feasibility of the inner-zonotope relaxation]\label{prop:membership}
$\wdem_k\in\Wmap(C_k)$ iff there exist generator coefficients $\xi_{\tau kh}$ with
\begin{equation}
\sum_{\tau,h}A_{kh}G_\tau\,\xi_{\tau kh}=\wdem_k,
\qquad
-z_{\tau kh}\mathbf 1\le\xi_{\tau kh}\le z_{\tau kh}\mathbf 1 .
\label{eq:tensor}
\end{equation}
Wrench feasibility of load $k$ is thus a linear feasibility problem in $(z,\xi)$, and
its projection onto $z$, $X_{\mathrm f}=\{z\in\mathcal X:H_w z\ge d\}$, is a polytope.
\end{lemma}
\begin{proof}
$z_{\tau kh}A_{kh}U_\tau^{\mathrm{in}}=\{A_{kh}G_\tau\xi:\|\xi\|_\infty\le z_{\tau kh}\}$;
the Minkowski sum superposes independent generators, giving \eqref{eq:tensor}, and
eliminating $\xi$ yields the polytope $X_{\mathrm f}$.
\end{proof}
By $U_\tau^{\mathrm{in}}\subseteq U_\tau(\phi)$ a certified wrench is feasible in this
heading-robust actuation-set model. Each mechanically attached base (grippers/tow bars) is
represented at the allocation layer by this conservative planar force set; we study
assignment and communication feasibility, not actuator-level realization, which is left to a
separate layer.

\subsection{Decision-induced robot-graph Laplacian $\Lap(z)$}
The information graph has a \emph{fixed} node set $\mathcal V=\{1,\dots,N\}$, one node per
robot \emph{identity}; every robot is always present, so no node is created, removed, or left
empty---an \emph{idle} robot still occupies its node and stays connected through the base
graph (the action set, of size $|\mathcal A|=MH{+}|\mathcal R|{+}1$, chooses \emph{edges} at a
node, not nodes). A fixed base graph $L_0\succeq0$ carries the always-on short-range
conductances; committing a type-$\tau$ robot to relay site $r$ \emph{gates on} a predefined
long-range link Laplacian $L_r\succeq0$ (effectiveness $\gamma_{\tau r}\ge0$), so
\begin{equation}
\Lap(z)=L_0+\sum_{\tau,r}\gamma_{\tau r}\,z_{\tau r}\,L_r\ \in\R^{N\times N}
\label{eq:laplacian}
\end{equation}
is affine in $z$ (Assumption~\ref{as:standing}(ii)) because \emph{only relay occupancy}
gates edges---not a general pairwise geometric graph, whose weights would be bilinear in
both endpoints' decisions. The individual variables are $x_{ir}\in\{0,1\}$ with
$z_{\tau r}=\sum_{i\in\mathcal N_\tau}x_{ir}$; a deterministic injective matching
$\chi_z:\{(\tau,r):z_{\tau r}\!=\!1\}\to\mathcal N_\tau$ (one robot per occupied site within a
type) instantiates $z$. \emph{Type-permutation invariance:} any two individual assignments
$x,x'$ with the same $z$ give $L(x')=\Pi^{\top}L(x)\Pi$ for a block-diagonal within-type
permutation $\Pi$, so $\lamtwo(\Lap(z))$ is well defined on the equivalence class. Each
$\Lap(z)$ is a symmetric Laplacian ($\Lap(z)\mathbf1=0$, $\Lap(z)\succeq0$), an affine
surrogate for the
physical $\Lgeo(q)$ on the \emph{same} $N$ robots: one robot per node, so $\Lap$ and $\Lgeo$
share the node set under the identity correspondence and no permutation is needed; the two
are compared through Lemma~\ref{lem:bridge}. All theorems use $\Lap(z)$.

Let $Q\in\R^{N\times(N-1)}$ be an orthonormal basis of $\mathbf 1^{\perp}$; the
information constraint is the LMI
\begin{equation}
Q^{\top}\Lap(z)\,Q\succeq\sigmastar I
\;\Longleftrightarrow\;
\lamtwo(\Lap(z))\ge\sigmastar,
\label{eq:lmi}
\end{equation}
affine in $z$, with PSD dual $Y\succeq0$; writing $Y=\pi Z$ ($\pi=\operatorname{tr}Y$,
$\operatorname{tr}Z=1$) separates the connectivity price $\pi$ from the modal direction
$Z$. The requirement $\sigmastar=\sigmadyn+\delta$ pairs the stability boundary $\sigmadyn$
(Remark~\ref{thm:stability}) with the margin $\delta$ of Lemma~\ref{lem:bridge}.

\begin{lemma}[Convex information set]\label{lem:convex}
$z\mapsto\lamtwo(\Lap(z))$ is concave (a pointwise minimum
$\min_{\|v\|=1,v\perp\mathbf1}v^{\top}\Lap(z)v$ of functions affine in
$z$~\cite{fiedler1973}), so \eqref{eq:lmi} defines a convex set~\cite{ghosh2006growing}.
\end{lemma}

\begin{lemma}[Geometric transfer under model mismatch]\label{lem:bridge}
Suppose $\|\Lgeo(q)-\Lap(z)\|_2\le\varepsilon_L$, the estimator obeys
$|\widehat\lamtwo-\lamtwo(\Lap(z))|\le\varepsilon_{\mathrm{est}}$, and the solver keeps
$\widehat\lamtwo\ge\sigmastar-\delta_{\mathrm{num}}$. Then Weyl's inequality gives
$\lamtwo(\Lgeo(q))\ge\lamtwo(\Lap(z))-\varepsilon_L$, so with
$\delta=\varepsilon_L+\varepsilon_{\mathrm{est}}+\delta_{\mathrm{num}}$ and
$\sigmastar:=\sigmadyn+\delta$ one obtains $\lamtwo(\Lgeo(q))\ge\sigmadyn$, the
optimizer--estimator (information-layer) stability threshold. For the exact centralized
SDP, $\Lap(z)$ is known and $\varepsilon_{\mathrm{est}}=\delta_{\mathrm{num}}=0$.
\end{lemma}
The convex theory of Sections~\ref{sec:wise}--\ref{sec:algorithm} uses $\Lap(z)$ and transfers
to $\Lgeo(q)$ only through Lemma~\ref{lem:bridge}. Its tracking part is \emph{a priori}: smooth
weights $w_{ij}=\psi(\|q_i-q_j\|)$ ($|\psi'|\le L_\psi$) and a tube $\|q_i-\bar q_i\|\le\rho$
give $\|\Lgeo(q)-\Lgeo(\bar q)\|_2\le4d_{\max}L_\psi\rho$; the residual surrogate term is
measured, and $\varepsilon_{\mathrm{est}},\delta_{\mathrm{num}}$ are fixed.
