\section{Population Model and Induced Network}
\label{sec:model}

\subsection{Assignment state (relaxed per-robot occupancy $z$)}
A team of $N=\sum_\tau n_\tau$ robots has $T$ types (heterogeneous in force and range),
$\tau(i)$ the type of robot $i$; there are $M$ loads with $H$ contact slots each, relay sites
$\mathcal R$, and the common action set
\begin{equation}
\mathcal A=\underbrace{\{(k,h)\}_{k\le M,\,h\le H}}_{\text{load--slot}}
\;\cup\;\mathcal R\;\cup\;\{\mathrm{o}\},
\label{eq:actions}
\end{equation}
with $\mathrm o$ idle; $a\in\mathcal A$ denotes a generic action, $(k,h)$ a load--slot,
$r$ a relay.

The \emph{assignment state} is the relaxed \emph{per-robot} occupancy matrix
$z=(z_{ia})\in\R_{\ge0}^{N\times|\mathcal A|}$: $z_{ia}\in[0,1]$ relaxes robot $i$'s single
integer action. Each robot spends a unit budget, and each slot or relay site is held by at
most one robot,
\begin{equation}
\begin{aligned}
&\textstyle\sum_{a\in\mathcal A} z_{ia}=1\ \ \forall i,
\qquad z_{ia}\ge0,\\[1mm]
&\textstyle\sum_i z_{i,kh}\le1\ \ \forall(k,h),
\qquad \sum_i z_{i,r}\le1\ \ \forall r.
\end{aligned}
\label{eq:conservation}
\end{equation}
The type aggregates $z_{\tau a}=\sum_{i\in\mathcal N_\tau}z_{ia}$ enter the productive
\emph{and} wrench layers (both couple through types); the communication layer uses the
per-robot relay occupancies $z_{ir}$ directly. The feasible set
$\mathcal X=\{z\in\R_{\ge0}^{N\times|\mathcal A|}:\eqref{eq:conservation}\}$ is a compact
convex polytope in $\R^{n}$, $n=N|\mathcal A|$ \emph{ambient} decision coordinates; the $N$
budget equalities reduce its affine dimension (to $N(|\mathcal A|-1)$ absent active facets),
and Theorem~\ref{thm:degeneracy} gives the face-specific dimension through the
active-constraint matrix. A finite team instantiates $z$ by rounding to
one action per robot (Remark~\ref{rem:integer}); the poses $q\in(\mathrm{SE}(2))^N$ enter
only the geometric instantiation (Section~\ref{sec:exp}).

\begin{assumption}[Standing hypotheses]\label{as:standing}
Throughout: \emph{(i)} $\mathcal X$ is compact convex and $\mathcal X_{\mathrm f}\neq
\varnothing$; \emph{(ii)} $\Lap(z)$ of \eqref{eq:laplacian} is affine in $z$ with each
$L_{ir}\succeq0$ (robot-indexed, fixed), symmetric; \emph{(iii)} $\nabla\lamtwo$ is used only
where $\lamtwo$ is simple, else the eigenspace directional derivative
(Prop.~\ref{prop:freeconn}); the SDP~\eqref{eq:lmi} stays exact regardless (no genericity
assumed); \emph{(iv)} the physical graph decomposes additively as
$\Lgeo(q,\hat z)=L_0^{\mathrm{geo}}(q)+\sum_{i,r}\hat z_{ir}L_{ir}^{\mathrm{geo}}(q)$ with
$L_0^{\mathrm{geo}}(q)\succeq L_0$ and $L_{ir}^{\mathrm{geo}}(q)\succeq L_{ir}$ for every
admissible pose (edge supports shared by several relays add; otherwise they are disjoint, so no
link is double-counted).
\end{assumption}

\subsection{Wrench feasibility}
A type-$\tau$ robot with heading $\phi$ exerts contact forces in the heading-aligned
ellipse $U_\tau(\phi)$ with semi-axes $F_\tau$, $\kappa F_\tau$ ($\kappa\le1$,
anisotropic actuation factor). Not tracking headings, we certify against the heading-independent \emph{inner zonotope}
$U_\tau^{\mathrm{in}}=\{G_\tau u:\|u\|_\infty\le1\}$ (generators $G_\tau$) inscribed in
$\mathrm{disk}(\kappa F_\tau)\subseteq U_\tau(\phi)$; feasibility tested against
$U_\tau^{\mathrm{in}}\subseteq U_\tau(\phi)$ is \emph{conservative}, with no false positive.

Let $A_{kh}\in\R^{3\times2}$ be the contact-to-wrench map at slot $(k,h)$ and
$\wdem_k\in\R^3$ the demanded wrench of load $k$. The reachable population wrench set is
the scaled Minkowski sum
$\Wmap(C_k)=\bigoplus_{\tau,h}z_{\tau kh}A_{kh}U_\tau^{\mathrm{in}}$.

\begin{lemma}[Exact feasibility of the inner-zonotope relaxation]\label{prop:membership}
$\wdem_k\in\Wmap(C_k)$ iff there exist generator coefficients $\xi_{\tau kh}$ with
\begin{equation}
\sum_{\tau,h}A_{kh}G_\tau\,\xi_{\tau kh}=\wdem_k,
\qquad
-z_{\tau kh}\mathbf 1\le\xi_{\tau kh}\le z_{\tau kh}\mathbf 1 .
\label{eq:tensor}
\end{equation}
Wrench feasibility of load $k$ is thus a linear feasibility problem in $(z,\xi)$, and
its projection onto $z$, $X_{\mathrm f}=\{z\in\mathcal X:H_w z\ge d\}$, is a polytope.
\end{lemma}
\begin{proof}[Proof sketch]
The scaled generators $z_{\tau kh}A_{kh}U_\tau^{\mathrm{in}}$ superpose (giving
\eqref{eq:tensor}); eliminating $\xi$ leaves the polytope $X_{\mathrm f}$.
\end{proof}
By $U_\tau^{\mathrm{in}}\subseteq U_\tau(\phi)$ a certified wrench is feasible in this
heading-robust model; we study assignment and communication feasibility, not actuator-level
realization.

\subsection{Decision-induced robot-graph Laplacian $\Lap(z)$}
The information graph has a \emph{fixed} node set $\mathcal V=\{1,\dots,N\}$, one node per
robot \emph{identity}: every robot is always present (an \emph{idle} robot still occupies its
node), and the action set chooses \emph{edges} at a node, not nodes. A fixed base graph
$L_0\succeq0$ carries the always-on short-range conductances (itself a lower-bound Laplacian);
when robot $i$ commits to relay site $r$ it \emph{gates on} its predefined long-range links
$L_{ir}=\gamma_{\tau(i)r}\sum_{j}\underline w^{r}_{ij}(e_i{-}e_j)(e_i{-}e_j)^{\top}\succeq0$,
weighted by the tube infima $\underline w^{r}_{ij}=\inf_{q\in\mathcal Q_{ir}}w_{ij}(q)$. The
tube $\mathcal Q_{ir}$---robot $i$ at relay $r$ and all \emph{admissible} positions of the
robots it reaches---is fixed before optimization, so each $L_{ir}$ is decision-independent and
the SDP stays affine (links of zero infimum weight are excluded, a more conservative backbone).
Both $L_0$ and every $L_{ir}$ are lower-bound Laplacians, giving $\Lgeo(q)\succeq\Lap(\hat z)$
(Assumption~\ref{as:standing}(iv), Lemma~\ref{lem:bridge}). Thus, \emph{directly in the
per-robot occupancy},
\begin{equation}
\Lap(z)=L_0+\sum_{i,r}\,z_{ir}\,L_{ir}\ \in\R^{N\times N}.
\label{eq:laplacian}
\end{equation}
Each $L_{ir}$ being fixed and robot-indexed makes $\Lap$ \emph{unambiguously affine} in $z$,
needing no matching or permutation (relay occupancy gates edges, not a bilinear pairwise
graph). $\Lap(z)$ is a symmetric Laplacian, an affine surrogate for the physical $\Lgeo(q)$ on
the \emph{same} $N$ robots (node $i$ is robot $i$), compared through Lemma~\ref{lem:bridge}.

Let $Q\in\R^{N\times(N-1)}$ be an orthonormal basis of $\mathbf 1^{\perp}$; the
information constraint is the LMI
\begin{equation}
Q^{\top}\Lap(z)\,Q\succeq\sigmastar I
\;\Longleftrightarrow\;
\lamtwo(\Lap(z))\ge\sigmastar,
\label{eq:lmi}
\end{equation}
affine in $z$, with PSD dual $Y\succeq0$; for $Y\neq0$, $\pi=\operatorname{tr}Y>0$ and
$Z=Y/\pi$ ($\operatorname{tr}Z=1$) separate the connectivity price $\pi$ from the modal
direction $Z$, while $Y=0$ ($\pi=0$) leaves $Z$ immaterial. The requirement
$\sigmastar>\sigmadyn$ is specified externally (Lemma~\ref{lem:bridge}).

\begin{lemma}[Convex information set]\label{lem:convex}
$z\mapsto\lamtwo(\Lap(z))$ is concave (a pointwise minimum
$\min_{\|v\|=1,v\perp\mathbf1}v^{\top}\Lap(z)v$ of functions affine in
$z$~\cite{fiedler1973}), so \eqref{eq:lmi} defines a convex set~\cite{ghosh2006growing}.
\end{lemma}

\begin{lemma}[A priori geometric transfer by graph monotonicity]\label{lem:bridge}
Let $\hat z\in\mathcal X\cap\{0,1\}^{n}$ be an integer recovery and $q$ any admissible pose. By
the additive decomposition of Assumption~\ref{as:standing}(iv) and the lower-bound
construction of $L_0,L_{ir}$,
\begin{equation}
\Lgeo(q)-\Lap(\hat z)=(L_0^{\mathrm{geo}}{-}L_0)+\textstyle\sum_{i,r}\hat z_{ir}
(L_{ir}^{\mathrm{geo}}{-}L_{ir})\succeq0,
\label{eq:mismatch}
\end{equation}
a sum of PSD terms with no link double-counted. Restricting to $\mathbf 1^{\perp}$, eigenvalue
monotonicity gives $\lamtwo(\Lgeo(q))\ge\lamtwo(\Lap(\hat z))$. Hence a certificate
$\lamtwo(\Lap(\hat z))\ge\sigmastar$ ($\sigmastar>\sigmadyn$ specified externally) yields
$\lamtwo(\Lgeo(q))\ge\sigmadyn$ \emph{a priori}, for every admissible pose---no post-hoc
mismatch term. Every certificate is re-evaluated in double precision, so solver tolerance does
not enter the guarantee; a future distributed estimator would add $\varepsilon_{\mathrm{est}}$,
zero here.
\end{lemma}
The convex theory of Sections~\ref{sec:wise}--\ref{sec:algorithm} uses $\Lap$ and transfers to
$\Lgeo$ through the Loewner bound \eqref{eq:mismatch} at the integer recovery $\hat z$.
