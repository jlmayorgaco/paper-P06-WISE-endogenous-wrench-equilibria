\section{Self-Sustaining Equilibria}
\label{sec:wise}

\subsection{Productive value and the optimal composition fiber}
The active load set is fixed: wrench feasibility ($z\in X_{\mathrm f}$) guarantees each
load its physical service, and the productive value allocates total served capacity through
the \emph{served-capacity} aggregate $y=Bz$, $y_k=\sum_{i,h}c_{\tau(i)} z_{ikh}$ (distinct
from the wrench aggregate $H_wz$; $y_k$ is the reserve for load $k$ beyond the wrench floor, a
proxy for admissible throughput):
\begin{equation}
V(z)=\phi(Bz),\qquad
\phi(y)=\sum_k\big(v_k\,y_k-\tfrac12\alpha\,y_k^2\big),
\label{eq:value}
\end{equation}
so $\phi(y)=-\tfrac12\|y-y_{\mathrm{ref}}\|_H^2+\mathrm{const}$ ($H=\alpha I\succ0$,
$y_{\mathrm{ref}}=v/\alpha$), strictly concave and pinning down a unique optimal aggregate (a
flat deficit penalty would leave it undetermined). This is a
constrained \emph{identical-interest aggregative game} in the per-robot occupancies, with
pseudo-gradient $[F(z)]_{ia}=-\partial V/\partial z_{ia}=-[B^{\top}\nabla\phi(Bz)]_{ia}$
(the population-game reading follows by grouping exchangeable same-type robots); the novelty
lies in what $V$ leaves undetermined.

Wrench feasibility is a \emph{hard} shared constraint through $X_{\mathrm f}$
(Lemma~\ref{prop:membership}), the projected matrix $H_w$ kept distinct from the objective
Hessian $H$.

\begin{definition}[Productive variational equilibrium]\label{def:ve}
A profile $z^{\star}\in X_{\mathrm f}$ is a \emph{productive variational equilibrium} if
$\langle F(z^{\star}),z-z^{\star}\rangle\ge0$ for all $z\in X_{\mathrm f}$, with $F=-\nabla V$.
As $V=\phi(Bz)$ is concave on the convex $X_{\mathrm f}$, this holds iff
$z^{\star}\in\arg\max_{X_{\mathrm f}}V$: the productive optimum \emph{is} the variational
equilibrium~\cite{facchinei2003fvi}, and its degeneracy is Proposition~\ref{prop:fibermono}.
\end{definition}

\begin{theorem}[Composition degeneracy]\label{thm:degeneracy}
Since $\phi$ is strictly concave on the compact convex $BX_{\mathrm f}$
($H=\alpha I\succ0$), the productively optimal aggregate $y^{\star}=Bz^{\star}$ is unique;
but the optimal set $\Emanifold=\{z\in X_{\mathrm f}:Bz=y^{\star}\}$ is a nonempty compact
convex polytope whose local dimension at a relative-interior point $\bar z$ is
\begin{equation}
\dim\Emanifold=\dim\!\big(\ker A\cap\ker B\cap\ker G_I\big)
=n-\operatorname{rank}[A;B;G_I],
\label{eq:dim}
\end{equation}
where $Az=b$ is budget conservation, $Bz=y^{\star}$ the served aggregate, $G_Iz=h_I$
the inequalities of $X_{\mathrm f}$ active at $\bar z$ (nonnegativity, occupancy, active
wrench), and $n=N|\mathcal A|$ is the ambient number of decision coordinates. Thus
$\Emanifold$ is positive-dimensional iff the
\emph{productive neutral space} $N_p(\bar z)=\ker A\cap\ker B\cap\ker G_I$ is nontrivial.
\end{theorem}
\begin{proof}[Proof sketch]
Uniqueness of $y^{\star}$ is strict concavity of $\phi$; at
$\bar z\in\operatorname{relint}\Emanifold$ the feasible perturbations are exactly $N_p(\bar z)$,
so $\Emanifold$ locally coincides with $\bar z+N_p(\bar z)$ and \eqref{eq:dim} is rank--nullity
(a general weighted $B$ can give $N_p=\{0\}$).
\end{proof}
For the two-load flagship ($n=72$), $\operatorname{rank}A=12$ and $\operatorname{rank}[A;B]=14$
with no inequality active at the interior optimum ($G_I=\varnothing$), so
$\dim\Emanifold=72-14=58$ (a single served aggregate would give $\dim=59$), certified by a
relative-interior point with residuals $\|Ad\|,\|Bd\|,\|G_Id\|<10^{-15}$. A boundary optimum
where wrench facets bind gives a smaller fiber.

\begin{proposition}[Fiber--monotonicity equivalence]\label{prop:fibermono}
Let $P_F$ have columns forming an orthonormal basis of the active-face tangent
$\mathcal T_F=\ker A\cap\ker G_I$. The pseudo-gradient
$F(z)=B^{\top}H(Bz-y_{\mathrm{ref}})$ obeys $\langle F(z{+}d)-F(z),d\rangle=\|H^{1/2}Bd\|^2$,
so its strong-monotonicity modulus on that face is
$\mu_F=\lambda_{\min}(P_F^{\top}B^{\top}HBP_F)$, and
$\mu_F=0\iff\ker(BP_F)\neq\{0\}\iff N_p(\bar z)\neq\{0\}$. Hence $\dim\Emanifold>0\iff\mu_F=0$: strong
monotonicity holds in the served aggregate ($H\succ0$) but is \emph{lost} on the
composition face---the degeneracy WISE exploits, unlike strongly contractive aggregative
games~\cite{martinezpiazuelo2022}.
\end{proposition}

A comparative-advantage exchange makes this concrete (Fig.~\ref{fig:flagship}): with
$c_\ell=2c_s$, moving one long lifter to the relay for two short lifters is neutral ($Bd=0$)
with $D\lamtwo[d]=\sum_{i,r}d_{ir}\,v^{\top}L_{ir}v$. Whether a \emph{free} gain exists is a
projection condition:

\begin{proposition}[Free-connectivity criterion]\label{prop:freeconn}
Let $\bar z\in\operatorname{relint}\Emanifold$ with $\lamtwo$ simple and Fiedler vector $v$,
so the locally feasible directions form the subspace $N_p=\ker A\cap\ker B\cap\ker G_I$ (no
constraint beyond those defining the minimal active face becomes newly active). With
$g_\lambda=\nabla_z\lamtwo(\Lap(\bar z))$ (entries $v^{\top}L_{ir}v$),
\begin{equation}
\max_{d\in N_p,\ \|d\|\le1}D\lamtwo(\bar z)[d]=\|\Pi_{N_p}g_\lambda\|,
\label{eq:freeconn}
\end{equation}
so a productively neutral improvement exists $\iff\Pi_{N_p}g_\lambda\neq0$, with optimal
direction $d^\star=\Pi_{N_p}g_\lambda/\|\Pi_{N_p}g_\lambda\|$. At a boundary $\bar z$ replace
$N_p$ by the tangent cone $T_{\Emanifold}(\bar z)=\{d:Ad{=}0,\ Bd{=}0,\ G_jd\le0,\ j\in
I(\bar z)\}$ ($I(\bar z)$ the active set); as $\Emanifold$ is convex, positivity of
\eqref{eq:freeconn} over $T_{\Emanifold}$ is equivalent to a local zero-cost improvement, and
none exists $\iff g_\lambda\in N_{\Emanifold}(\bar z)=\{A^{\top}\alpha+B^{\top}\beta+
G_I^{\top}\nu:\nu\ge0\}$, the KKT normal cone. Under a repeated $\lamtwo$,
replace $D\lamtwo[d]$ by $\lambda_{\min}(U_2^{\top}D\Lap(\bar z)[d]\,U_2)$ with $U_2$ an
orthonormal basis of the smallest-eigenvalue eigenspace of $Q^{\top}\Lap(\bar z)Q$; a neutral
improvement then exists
$\iff\max_{d\in T_{\Emanifold},\|d\|\le1}\lambda_{\min}(U_2^{\top}D\Lap[d]U_2)>0$ (a small
SDP).
\end{proposition}

\subsection{The WISE refinement}
\begin{definition}[WISE and its integer counterparts]\label{def:wise}
A profile $z^{\star}\in\Emanifold$ is a \emph{Wrench- and
Information-Self-Sustaining Equilibrium} if $\lamtwo(\Lap(z^{\star}))\ge\sigmastar$, i.e.\
$z^{\star}\in\Essrel:=\Emanifold\cap\Cdelta$ over the fractional occupancy relaxation, where
$\Cdelta=\{z\in X_{\mathrm f}:\lamtwo(\Lap(z))\ge\sigmastar\}$. For finite teams three
integer notions differ: the \emph{self-sustaining feasible} set
$\mathcal F_{\mathbb Z}=\{z\in X_{\mathrm f}\cap\{0,1\}^{n}:\lamtwo(\Lap(z))\ge\sigmastar\}$
(wrench- and connectivity-feasible), its \emph{optimum}
$V_{\mathbb Z}^{\star}=\max_{\mathcal F_{\mathbb Z}}V$, and the \emph{zero-cost integer WISE}
$\Essint=\mathcal F_{\mathbb Z}\cap\{z:Bz=y^{\star}\}$, nonempty iff
$V_{\mathbb Z}^{\star}=V^{\star}$; the productive integrality gap is
$g_{\mathbb Z}=V^{\star}-V_{\mathbb Z}^{\star}\ge0$. \emph{Zero-cost} means exact aggregate
preservation $B\hat z=y^{\star}$; numerically it is declared when
$\|B\hat z-y^{\star}\|_\infty\le\tau_B$ ($\tau_B=10^{-8}$), an implementation criterion that
does not alter this definition. An integer self-sustaining assignment need not be zero-cost. Since
$\sigmastar>\sigmadyn$, the affine certificate clears the information-layer threshold by the
explicit margin $\sigmastar-\sigmadyn$ (Remark~\ref{thm:stability}), and Lemma~\ref{lem:bridge}
transfers this to every admissible geometric realization; $\Cdelta$ is convex and closed
(Lemma~\ref{lem:convex}). WISE is a \emph{lexicographic selection} from the productive VE set,
not an equilibrium of the productive game alone.
\end{definition}

\begin{theorem}[Relaxed WISE existence and selection as an SDP]\label{thm:existence}
Over the fractional occupancy relaxation, \emph{Stage 1} fixes the unique optimal aggregate
$y^{\star}=Bz^{\star}$, $z^{\star}\in\arg\max_{z\in X_{\mathrm f}}V(z)$
(Theorem~\ref{thm:degeneracy}); \emph{Stage 2} selects connectivity on the resulting
fiber. The selector $z^{\star}_{\mathrm{WISE}}\in
\arg\max_{z\in\Emanifold}\lamtwo(\Lap(z))$ solves the Stage-2 SDP in the \emph{lifted} wrench
variables (Lemma~\ref{prop:membership}, no facet enumeration of $X_{\mathrm f}$),
\begin{equation}
\LamE=\max_{z,\,\xi,\,t}\ t\quad\text{s.t.}\quad
z\in\mathcal X,\ Bz=y^{\star},\ \eqref{eq:tensor},\ Q^{\top}\Lap(z)\,Q\succeq tI,
\label{eq:sdp}
\end{equation}
convex by Lemma~\ref{lem:convex} (affine constraints in $(z,\xi)$, concave
$\lamtwo\!\circ\!\Lap$). Over the relaxation a \emph{relaxed} WISE exists iff the value
clears the requirement,
\begin{equation}
\Essrel\neq\varnothing\iff\LamE\ge\sigmastar,
\label{eq:exist}
\end{equation}
and any maximiser $z^{\star}_{\mathrm{WISE}}$ preserves the productive optimum exactly,
$Bz^{\star}_{\mathrm{WISE}}=y^{\star}$, hence is itself a variational equilibrium
(Def.~\ref{def:ve}) with $V=V^{\star}$. When
$\min_{\Emanifold}\lamtwo<\sigmastar\le\LamE$ the refinement is \emph{strict}. If
$\Emanifold^{\star}$ is non-singleton, the strongly convex tie-break
$\min_{\Emanifold^{\star}}\tfrac12\|z-z_{\mathrm{prev}}\|_W^2$ ($W\succ0$, least role change)
fixes a \emph{unique} selector.
\end{theorem}
\begin{proof}[Proof sketch]
The constraints are convex (Lemma~\ref{lem:convex}), so \eqref{eq:sdp} is an SDP with value
$\LamE$ attained on the compact $\Emanifold$; $\Essrel\neq\varnothing$ iff $\LamE\ge\sigmastar$,
and feasibility forces $Bz^{\star}_{\mathrm{WISE}}=y^{\star}$.
\end{proof}

\subsection{Aggregate--estimator stability (motivating)}
\begin{remark}[Connectivity threshold]\label{thm:stability}
A standard small-gain model---a served-aggregate error coupled to a consensus estimator over
$Q^{\top}\Lap(z^{\star})Q$ with gains $\mF,\vt_1,\vt_2>0$---is exponentially stable when
$\lamtwo>\sigmadyn:=\vt_1\vt_2/(c\,\mF)$. This motivates treating $\sigmadyn>0$ as an external
information-layer requirement; the selector enforces $\sigmastar>\sigmadyn$ and
Lemma~\ref{lem:bridge} transfers it to the geometric graph a priori. No online realization is
analyzed here.
\end{remark}

\subsection{The price of self-sustainability}
\label{sec:price}
The lexicographic selector of Theorem~\ref{thm:existence} is exact; a scalarised
$V_\varepsilon=V+\varepsilon\lamtwo$ recovers it only as $\varepsilon\to0$
(Tikhonov~\cite{facchinei2003fvi}: $\|Bz_\varepsilon-y^{\star}\|\le
\sqrt{2\varepsilon\operatorname{osc}(\lamtwo)/\alpha}$), so a finite $\varepsilon$ moves the
aggregate---a Pareto scalarisation, not the lexicographic form.

Beyond the free fiber the requirement carries a productive cost. Define the
\emph{price of self-sustainability}
\begin{equation}
\Price(\sigma)=V^{\star}-\max\{V(z):z\in X_{\mathrm f},\ \lamtwo(\Lap(z))\ge\sigma\}
\label{eq:price}
\end{equation}
($\Price(\sigma)=+\infty$ when the set is empty), and the two spectral capacities
$\LamE=\max_{z\in\Emanifold}\lamtwo(\Lap(z))$ and
$\LamX=\max_{z\in X_{\mathrm f}}\lamtwo(\Lap(z))\ge\LamE$.

\begin{theorem}[Price and the free/costly/impossible trichotomy]\label{thm:price}
$\Price$ is a nondecreasing, proper extended-value convex function, and
\begin{equation}
\Price(\sigma)=
\begin{cases}
0, & \sigma\le\LamE\ \ (\text{free}),\\[1pt]
\text{finite}>0, & \LamE<\sigma\le\LamX\ \ (\text{costly}),\\[1pt]
+\infty, & \sigma>\LamX\ \ (\text{impossible}).
\end{cases}
\label{eq:trichotomy}
\end{equation}
At any $\sigma<\LamX$ admitting a strictly feasible point ($\exists z\in X_{\mathrm f}$ with
$\lamtwo(\Lap(z))>\sigma$, i.e.\ Slater holds), the optimal multiplier $\pi^{\star}$ of the
connectivity constraint satisfies $\pi^{\star}\in\partial\Price(\sigma)$: the marginal
productive price of one unit of connectivity, in the units of $V$.
\end{theorem}
\begin{proof}[Proof sketch]
The super-level set (Lemma~\ref{lem:convex}) shrinks in $\sigma$, so $\Price$ is nondecreasing;
$\sigma$ enters linearly, so $\Price$ is a convex perturbation value with subdifferential the
optimal multiplier set under Slater~\cite{boyd2004convex}. The cases follow from
$\sigma\le\LamE$ ($\Price=0$), $\sigma>\LamX$ (infeasible), and between.
\end{proof}

\subsection{Finite-team recovery}
A sufficient margin test preserves \emph{joint feasibility} after integer recovery (it does
not by itself preserve the aggregate). Let $z^{\star}$ be a relaxed WISE and define the
wrench and connectivity slacks and the sensitivity constants
\begin{equation}
\begin{aligned}
&m_w=\min_j\big((H_wz^{\star})_j-d_j\big),\quad
m_\lambda=\lamtwo(\Lap(z^{\star}))-\sigmastar,\\
&\kappa_w=\|H_w\|_{2\to\infty},\qquad
\kappa_L=\sup_{\|d\|=1}\|D\Lap[d]\|_2 .
\end{aligned}
\label{eq:margins}
\end{equation}
\begin{proposition}[Margin-based preservation of joint feasibility]\label{prop:integer}
Let $\hat z$ be any integer assignment meeting budgets, integrality and occupancy exactly
with $\|\hat z-z^{\star}\|_2\le r$. If
\begin{equation}
r<\min\!\Big\{\tfrac{m_w}{\kappa_w},\ \tfrac{m_\lambda}{\kappa_L}\Big\},
\label{eq:intmargin}
\end{equation}
then $\hat z\in X_{\mathrm f}$ and $\lamtwo(\Lap(\hat z))\ge\sigmastar$: $\hat z$ is an integer
self-sustaining assignment. If additionally $B\hat z=y^{\star}$ it is a zero-cost integer WISE;
otherwise it has \emph{recovery loss} $\ell(\hat z)=V^{\star}-V(\hat z)$. The optimal
integrality gap $g_{\mathbb Z}=V^{\star}-V_{\mathbb Z}^{\star}$ is a distinct quantity,
evaluated by the exact oracle (Section~\ref{sec:exp}).
\end{proposition}
\begin{proof}[Proof sketch]
Row-wise $(H_w\hat z)_j\ge d_j+m_w-\kappa_w r>d_j$ gives $\hat z\in X_{\mathrm f}$; and, as
$\Lap$ is affine, Weyl's inequality gives $\lamtwo(\Lap(\hat z))\ge\sigmastar+m_\lambda-
\kappa_L r>\sigmastar$.
\end{proof}

\begin{remark}[Rounding as a search]\label{rem:integer}
Proposition~\ref{prop:integer} is a checkable sufficient condition, not an existence
guarantee: best-of-$R$ rounding \emph{searches} for a feasible $\hat z$ and \emph{directly}
re-checks it against budgets, occupancy, the lifted wrench LP and $\lamtwo\ge\sigmastar$ (no
$O(1/N)$ claim). The radius~\eqref{eq:intmargin} is a separate, conservative certificate,
dominated by the spectral term $m_\lambda/\kappa_L$ (Cauchy--Schwarz $\kappa_L$;
$\chi{\approx}190$ on the instances of Section~\ref{sec:exp}), so direct re-certification is
the operative test.
\end{remark}
