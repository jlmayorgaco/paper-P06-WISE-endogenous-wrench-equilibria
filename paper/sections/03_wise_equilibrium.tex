\section{Self-Sustaining Equilibria}
\label{sec:wise}

\subsection{Productive value and the optimal composition fiber}
The active load set is fixed: wrench feasibility ($z\in X_{\mathrm f}$) guarantees each
load its physical service, and the productive value allocates total served capacity through
the \emph{served-capacity} aggregate $y=Bz$, $y_k=\sum_{i,h}c_{\tau(i)} z_{ikh}$ (distinct
from the wrench aggregate $H_wz$; $y_k$ is the reserve for load $k$ beyond the wrench floor, a
proxy for admissible throughput):
\begin{equation}
V(z)=\phi(Bz),\qquad
\phi(y)=\sum_k\big(v_k\,y_k-\tfrac12\alpha\,y_k^2\big),
\label{eq:value}
\end{equation}
so $\phi(y)=-\tfrac12\|y-y_{\mathrm{ref}}\|_H^2+\mathrm{const}$ ($H=\alpha I\succ0$,
$y_{\mathrm{ref}}=v/\alpha$), strictly concave and pinning down a unique optimal aggregate (a
flat deficit penalty would leave it undetermined). This is a
constrained \emph{identical-interest aggregative game} in the per-robot occupancies, with
pseudo-gradient $[F(z)]_{ia}=-\partial V/\partial z_{ia}=-[B^{\top}\nabla\phi(Bz)]_{ia}$
(the population-game reading follows by grouping exchangeable same-type robots); the novelty
lies in what $V$ leaves undetermined.

Wrench feasibility is a \emph{hard} shared constraint through $X_{\mathrm f}$
(Lemma~\ref{prop:membership}), the projected matrix $H_w$ kept distinct from the objective
Hessian $H$.

\begin{definition}[Productive variational equilibrium]\label{def:ve}
A profile $z^{\star}\in X_{\mathrm f}$ is a \emph{productive variational equilibrium} if
$\langle F(z^{\star}),z-z^{\star}\rangle\ge0$ for all $z\in X_{\mathrm f}$, with $F=-\nabla V$.
As $V=\phi(Bz)$ is concave on the convex $X_{\mathrm f}$, this holds iff
$z^{\star}\in\arg\max_{X_{\mathrm f}}V$, i.e.\ the solution \emph{set} coincides with the
productive optimal set, $\operatorname{SOL}(X_{\mathrm f},F)=\arg\max_{X_{\mathrm f}}V
=\Emanifold$~\cite{facchinei2003fvi}. The aggregate $y^{\star}$ is unique but this set need not
be a singleton; its degeneracy is Proposition~\ref{prop:fibermono}.
\end{definition}

\begin{theorem}[Composition degeneracy]\label{thm:degeneracy}
Since $\phi$ is strictly concave on the compact convex $BX_{\mathrm f}$
($H=\alpha I\succ0$), the productively optimal aggregate $y^{\star}=Bz^{\star}$ is unique;
but the optimal set $\Emanifold=\{z\in X_{\mathrm f}:Bz=y^{\star}\}$ is a nonempty compact
convex polytope whose local dimension at a relative-interior point $\bar z$ is
\begin{equation}
\dim\Emanifold=\dim\!\big(\ker A\cap\ker B\cap\ker G_I\big)
=n-\operatorname{rank}[A;B;G_I],
\label{eq:dim}
\end{equation}
where $Az=b$ is budget conservation, $Bz=y^{\star}$ the served aggregate, $G_Iz=h_I$
the inequalities of $X_{\mathrm f}$ active at $\bar z$ (nonnegativity, occupancy, active
wrench), and $n=N|\mathcal A|$ is the ambient number of decision coordinates. Thus
$\Emanifold$ is positive-dimensional iff the
\emph{productive neutral space} $N_p(\bar z)=\ker A\cap\ker B\cap\ker G_I$ is nontrivial.
\end{theorem}
\begin{proof}[Proof sketch]
Uniqueness of $y^{\star}$ is strict concavity of $\phi$; at
$\bar z\in\operatorname{relint}\Emanifold$ the feasible perturbations are exactly $N_p(\bar z)$,
so $\Emanifold$ locally coincides with $\bar z+N_p(\bar z)$ and \eqref{eq:dim} is rank--nullity
(a general weighted $B$ can give $N_p=\{0\}$).
\end{proof}
For the two-load flagship ($n=72$), $\operatorname{rank}A=12$ and $\operatorname{rank}[A;B]=14$
with no inequality active at the interior optimum ($G_I=\varnothing$), so
$\dim\Emanifold=72-14=58$ (a single served aggregate would give $\dim=59$), certified by a
relative-interior point with residuals $\|Ad\|,\|Bd\|,\|G_Id\|<10^{-15}$. A boundary optimum
where wrench facets bind gives a smaller fiber.

\begin{proposition}[Fiber--monotonicity equivalence]\label{prop:fibermono}
Let $P_F$ have columns forming an orthonormal basis of the active-face tangent
$\mathcal T_F=\ker A\cap\ker G_I$, assumed nontrivial ($\dim\mathcal T_F>0$; if
$\mathcal T_F=\{0\}$ the face is a singleton, $\Emanifold=\{\bar z\}$ is trivially
zero-dimensional and no selection remains). The pseudo-gradient
$F(z)=B^{\top}H(Bz-y_{\mathrm{ref}})$ obeys $\langle F(z{+}d)-F(z),d\rangle=\|H^{1/2}Bd\|^2$,
so its strong-monotonicity modulus on that face is
$\mu_F=\lambda_{\min}(P_F^{\top}B^{\top}HBP_F)$, and
$\mu_F=0\iff\ker(BP_F)\neq\{0\}\iff N_p(\bar z)\neq\{0\}$. Hence $\dim\Emanifold>0\iff\mu_F=0$: strong
monotonicity holds in the served aggregate ($H\succ0$) but is \emph{lost} on the
composition face---the degeneracy WISE exploits, unlike strongly contractive aggregative
games~\cite{martinezpiazuelo2022}.
\end{proposition}

A comparative-advantage exchange makes this concrete (Fig.~\ref{fig:flagship}): with
$c_\ell=2c_s$, moving one long lifter to the relay for two short lifters is neutral ($Bd=0$)
while $\lamtwo$ strictly rises. Whether such a \emph{free} gain exists---and whether it has
been exhausted---is decided by a single scalar.

Orient every inequality of $X_{\mathrm f}$ uniformly as $Gz\ge h$ (negate occupancy caps
$\sum_iz_{i,kh}\le1$; nonnegativity and $H_wz\ge d$ are already in this form), and write
$G_I$ for the rows active at $\bar z$.

\begin{definition}[Free-networkability modulus]\label{def:gamma}
For $\bar z\in\Emanifold$ let $T_{\Emanifold}(\bar z)=\{d:Ad{=}0,\ Bd{=}0,\ G_Id\ge0\}$ be the
tangent cone. Let $\widetilde L(\bar z)=Q^{\top}\Lap(\bar z)Q$ and let $\widetilde U_2$ have
orthonormal columns spanning the \emph{minimum}-eigenvalue eigenspace of $\widetilde L(\bar z)$
(note $\lambda_{\min}(\widetilde L)=\lamtwo(\Lap)$); set
\begin{equation}
U_2(\bar z)=Q\,\widetilde U_2\in\R^{N\times m},\qquad
m=\dim\ker\!\big(\widetilde L(\bar z)-\lamtwo I\big),
\label{eq:u2}
\end{equation}
so $U_2^{\top}\mathbf1=0$ \emph{by construction}. Passing through $Q$ is not cosmetic: on a
disconnected graph $\lamtwo(\Lap)=0$ is degenerate \emph{together with} the consensus mode, and
a basis of $\ker\Lap$ need not avoid $\mathbf1$; $\widetilde L$ removes $\mathbf1$ structurally.
We call $\lamtwo$ \emph{simple} when $m=1$ \emph{for $\widetilde L$}. As $\Lap$ is affine,
$D\Lap[d]=\sum_{i,r}d_{ir}L_{ir}$ and the directional derivative is the spectral-function
formula~\cite{lewis1996spectral,overton1992large}
\begin{equation}
D\lamtwo(\bar z)[d]=\lambda_{\min}\!\big(U_2^{\top}D\Lap[d]\,U_2\big),
\label{eq:dirderiv}
\end{equation}
which reduces to $\langle g_\lambda,d\rangle$, $[g_\lambda]_{ir}=v^{\top}L_{ir}v$, when
$m=1$ with Fiedler vector $v=U_2$. The \emph{free-networkability modulus} is
\begin{equation}
\Gamma_{\Emanifold}(\bar z)=\max\big\{D\lamtwo(\bar z)[d]:\
d\in T_{\Emanifold}(\bar z),\ \|d\|_2\le1\big\}.
\label{eq:gamma}
\end{equation}
\end{definition}

\begin{theorem}[Optimal-fiber networkability: a local test certifies global optimality]
\label{thm:networkability}
Let $\bar z\in\Emanifold$, with $\Emanifold$ the compact convex \emph{polyhedral} fiber of
Theorem~\ref{thm:degeneracy}. Then $\Gamma_{\Emanifold}(\bar z)\ge0$ and
\begin{equation}
\Gamma_{\Emanifold}(\bar z)>0
\iff
\exists\,z'\in\Emanifold:\ \lamtwo(\Lap(z'))>\lamtwo(\Lap(\bar z)),
\label{eq:iff}
\end{equation}
equivalently $\Gamma_{\Emanifold}(\bar z)=0\iff\bar z\in
\arg\max_{z\in\Emanifold}\lamtwo(\Lap(z))$, i.e.\ $\lamtwo(\Lap(\bar z))=\LamE$. The
\emph{local} directional test therefore certifies \emph{global} connectivity optimality over
the entire fiber, and quantifies the remaining gap,
\begin{equation}
\LamE-\lamtwo(\Lap(\bar z))\ \le\
\Gamma_{\Emanifold}(\bar z)\,\|z^{\star}_{\mathrm{WISE}}-\bar z\|_2 .
\label{eq:gammabound}
\end{equation}
At $\bar z\in\operatorname{relint}\Emanifold$ with $\lamtwo$ simple, $T_{\Emanifold}=N_p$ and
\eqref{eq:gamma} is available in closed form,
$\Gamma_{\Emanifold}(\bar z)=\|\Pi_{N_p}g_\lambda\|_2$, with optimal direction
$d^{\star}=\Pi_{N_p}g_\lambda/\|\Pi_{N_p}g_\lambda\|_2$; no free gain exists $\iff
g_\lambda\in N_{\Emanifold}(\bar z)=\{A^{\top}\alpha+B^{\top}\beta+G_I^{\top}\nu:\nu\le0\}$,
the KKT normal cone polar to $T_{\Emanifold}(\bar z)$ under the $Gz\ge h$ orientation. In
general \eqref{eq:gamma} is a small SDP,
$\max\{t:d\in T_{\Emanifold}(\bar z),\ \|d\|_2\le1,\ U_2^{\top}D\Lap[d]U_2\succeq tI\}$,
convex since $d\mapsto\lambda_{\min}(U_2^{\top}D\Lap[d]U_2)$ is concave.
\end{theorem}
\begin{proof}[Proof sketch]
$\Gamma_{\Emanifold}\ge0$ since $d=0$ is admissible. By Lemma~\ref{lem:convex},
$f=\lamtwo\circ\Lap$ is finite and concave, so $Df(\bar z)[\cdot]$ exists, is positively
homogeneous, and overestimates increments: $f(z')-f(\bar z)\le Df(\bar z)[z'-\bar z]$ for all
$z'$; \eqref{eq:dirderiv} is the standard directional derivative of a minimum eigenvalue over
its eigenspace. ($\Leftarrow$) If $z'\in\Emanifold$ improves $f$, then
$d_0=z'-\bar z\in T_{\Emanifold}(\bar z)$ satisfies $Df(\bar z)[d_0]\ge f(z')-f(\bar z)>0$,
and homogeneity rescales to $\|d_0\|_2\le1$. ($\Rightarrow$) If $Df(\bar z)[d]>0$ for some
$d\in T_{\Emanifold}(\bar z)$, polyhedrality gives
$T_{\Emanifold}(\bar z)=\operatorname{cone}(\Emanifold-\bar z)$ (already closed), so
$\bar z+t_0d\in\Emanifold$ for some $t_0>0$; by definition of the directional derivative
$f(\bar z+td)>f(\bar z)$ for all small $t\in(0,t_0]$, and $\bar z+td\in\Emanifold$ by
convexity. The second equivalence is the contrapositive. For \eqref{eq:gammabound} put
$\delta=z^{\star}_{\mathrm{WISE}}-\bar z$; concavity gives
$\LamE-f(\bar z)\le Df(\bar z)[\delta]$, positive homogeneity gives
$Df(\bar z)[\delta]=\|\delta\|_2\,Df(\bar z)[\delta/\|\delta\|_2]$, and
$\delta/\|\delta\|_2$ is a unit vector of $T_{\Emanifold}(\bar z)$, so the last factor is at
most $\Gamma_{\Emanifold}(\bar z)$ by definition~\eqref{eq:gamma}. Since
$\|\delta\|_2\le\operatorname{diam}\Emanifold$, the pose-independent bound
$\LamE-f(\bar z)\le\Gamma_{\Emanifold}(\bar z)\operatorname{diam}\Emanifold$ also holds.
\end{proof}

\begin{remark}[Degeneracy versus \emph{network-visible} degeneracy]\label{rem:dnet}
$\dim\Emanifold$ counts productively neutral directions, but only those the communication
layer can see at all are a candidate design resource. Define the \emph{network-visible neutral
dimension} $d_{\mathrm{net}}=\operatorname{rank}\big(D\Lap|_{N_p}\big)$; then
\begin{equation}
d_{\mathrm{net}}\le\min\big\{\dim N_p,\ N|\mathcal R|,\
\dim\operatorname{span}\{L_{ir}\}_{i,r}\big\},
\label{eq:dnet}
\end{equation}
since only the $N|\mathcal R|$ relay coordinates enter $\Lap$ and its range lies in the span of
the $L_{ir}$. Visibility is necessary but not sufficient for a gain \emph{at $\bar z$}: a
direction can move $\Lap$ without moving the active Fiedler mode, so
$\Gamma_{\Emanifold}(\bar z)>0\Rightarrow d_{\mathrm{net}}>0$ but not conversely---%
$\Gamma_{\Emanifold}$, not $d_{\mathrm{net}}$, is the usability test. Empirically the two
scales differ by an order of magnitude (E1b): $\dim\Emanifold=59$ while
$d_{\mathrm{net}}\in[10,12]$ on those single-aggregate instances ($|\mathcal R|=1$, so
$d_{\mathrm{net}}\le N=12$ binds); the two-load flagship has $\dim\Emanifold=58$.
\end{remark}

\subsection{The WISE refinement}
\begin{definition}[WISE and its integer counterparts]\label{def:wise}
A profile $z^{\star}\in\Emanifold$ is a \emph{Wrench- and
Information-Self-Sustaining Equilibrium} if $\lamtwo(\Lap(z^{\star}))\ge\sigmastar$, i.e.\
$z^{\star}\in\Essrel:=\Emanifold\cap\Cdelta$ over the fractional occupancy relaxation, where
$\Cdelta=\{z\in X_{\mathrm f}:\lamtwo(\Lap(z))\ge\sigmastar\}$. For finite teams three
integer notions differ: the \emph{self-sustaining feasible} set
$\mathcal F_{\mathbb Z}=\{z\in X_{\mathrm f}\cap\{0,1\}^{n}:\lamtwo(\Lap(z))\ge\sigmastar\}$
(wrench- \emph{and} connectivity-feasible), its \emph{optimum}
$V_{\mathbb Z}^{\mathrm{ss}}=\max_{\mathcal F_{\mathbb Z}}V$, and the \emph{zero-cost integer
WISE} $\Essint=\mathcal F_{\mathbb Z}\cap\{z:Bz=y^{\star}\}$, nonempty iff
$V_{\mathbb Z}^{\mathrm{ss}}=V^{\star}$. Since $\mathcal F_{\mathbb Z}$ imposes connectivity as
well as integrality, $g_{\mathbb Z}^{\mathrm{ss}}=V^{\star}-V_{\mathbb Z}^{\mathrm{ss}}\ge0$ is
the \emph{integer self-sustaining gap}, not a pure productive integrality gap; isolating the
latter needs $V_{\mathbb Z}^{\mathrm{prod}}=\max_{X_{\mathrm f}\cap\{0,1\}^n}V$, which we do
not compute here. \emph{Zero-cost} means exact aggregate
preservation $B\hat z=y^{\star}$; numerically it is declared when
$\|B\hat z-y^{\star}\|_\infty\le\tau_B$ ($\tau_B=10^{-8}$), an implementation criterion that
does not alter this definition. An integer self-sustaining assignment need not be zero-cost. Since
$\sigmastar>\sigmadyn$, any \emph{integer} member of $\mathcal F_{\mathbb Z}$ clears the
information-layer threshold by the explicit margin $\sigmastar-\sigmadyn$
(Prop.~\ref{thm:stability}), and Lemma~\ref{lem:bridge} then transfers it to every admissible
geometric realization of that assignment; $\Cdelta$ is convex and closed
(Lemma~\ref{lem:convex}). Every WISE \emph{is} a productive variational equilibrium;
connectivity acts as a \emph{lexicographic refinement} of the set-valued solution
$\operatorname{SOL}(X_{\mathrm f},F)=\Emanifold$, not as an additional strategic equilibrium
condition. Equivalently, $z^{\star}_{\mathrm{WISE}}$ satisfies the two-level condition
\begin{equation}
0\in\partial\big[-\lamtwo(\Lap(z^{\star}_{\mathrm{WISE}}))\big]+N_{\Emanifold}
(z^{\star}_{\mathrm{WISE}}),\qquad \Emanifold=\operatorname{SOL}(X_{\mathrm f},F),
\label{eq:hier}
\end{equation}
whose primal test is exactly $\Gamma_{\Emanifold}=0$ (Theorem~\ref{thm:networkability}). No
distributed equilibrium-seeking dynamics are claimed.
\end{definition}

\begin{theorem}[Relaxed WISE existence and selection as an SDP]\label{thm:existence}
Over the fractional occupancy relaxation, \emph{Stage 1} fixes the unique optimal aggregate
$y^{\star}=Bz^{\star}$, $z^{\star}\in\arg\max_{z\in X_{\mathrm f}}V(z)$
(Theorem~\ref{thm:degeneracy}); \emph{Stage 2} selects connectivity on the resulting
fiber. The selector $z^{\star}_{\mathrm{WISE}}\in
\arg\max_{z\in\Emanifold}\lamtwo(\Lap(z))$ solves the Stage-2 SDP in the \emph{lifted} wrench
variables (Lemma~\ref{prop:membership}, no facet enumeration of $X_{\mathrm f}$),
\begin{equation}
\LamE=\max_{z,\,\xi,\,t}\ t\quad\text{s.t.}\quad
z\in\mathcal X,\ Bz=y^{\star},\ \eqref{eq:tensor},\ Q^{\top}\Lap(z)\,Q\succeq tI,
\label{eq:sdp}
\end{equation}
convex by Lemma~\ref{lem:convex} (affine constraints in $(z,\xi)$, concave
$\lamtwo\!\circ\!\Lap$). Over the relaxation a \emph{relaxed} WISE exists iff the value
clears the requirement,
\begin{equation}
\Essrel\neq\varnothing\iff\LamE\ge\sigmastar,
\label{eq:exist}
\end{equation}
and any maximiser $z^{\star}_{\mathrm{WISE}}$ preserves the productive optimum exactly,
$Bz^{\star}_{\mathrm{WISE}}=y^{\star}$, hence is itself a variational equilibrium
(Def.~\ref{def:ve}) with $V=V^{\star}$. When
$\min_{\Emanifold}\lamtwo<\sigmastar\le\LamE$ the refinement is \emph{strict}. If the
connectivity-optimal face $\Emanifold^{\star}:=\arg\max_{z\in\Emanifold}\lamtwo(\Lap(z))$
(equivalently $\{z\in\Emanifold:\Gamma_{\Emanifold}(z)=0\}$ by
Theorem~\ref{thm:networkability}) is non-singleton, the strongly convex tie-break
$\min_{\Emanifold^{\star}}\tfrac12\|z-z_{\mathrm{prev}}\|_W^2$ ($W\succ0$, least role change)
fixes a \emph{unique} selector.
\end{theorem}
\begin{proof}[Proof sketch]
The constraints are convex (Lemma~\ref{lem:convex}), so \eqref{eq:sdp} is an SDP with value
$\LamE$ attained on the compact $\Emanifold$; $\Essrel\neq\varnothing$ iff $\LamE\ge\sigmastar$,
and feasibility forces $Bz^{\star}_{\mathrm{WISE}}=y^{\star}$.
\end{proof}

\subsection{What connectivity buys: information-layer stability}
\begin{proposition}[Threshold and guaranteed rate]\label{thm:stability}
Consider the \emph{reduced information-layer interconnection} in the coordinates
$a,b\in\R^{N-1}$ of $\mathbf1^{\perp}$: a served-aggregate coordination error $a$ driven by a
distributed consensus estimator error $b$ diffusing over $\widetilde L(z)=Q^{\top}\Lap(z)Q$,
\begin{equation}
\dot a=-\mF\,a+\vt_1 b,\qquad
\dot b=\vt_2\,a-c\,\widetilde L(z)\,b,
\label{eq:interconnection}
\end{equation}
with damping $\mF>0$, consensus gain $c>0$ and \emph{isotropic, mode-aligned} cross-couplings
$\vt_1,\vt_2>0$ (i.e.\ $\vt_1I,\vt_2I$), so that \eqref{eq:interconnection} block-diagonalises
in the eigenbasis of $\widetilde L$. Each eigenvalue $\lambda$ of $\widetilde L$ then
contributes an independent scalar pair
\begin{equation}
\begin{bmatrix}\dot a_\lambda\\ \dot b_\lambda\end{bmatrix}
=A_\lambda\begin{bmatrix}a_\lambda\\ b_\lambda\end{bmatrix},\qquad
A_\lambda=\begin{bmatrix}-\mF&\vt_1\\ \vt_2&-c\lambda\end{bmatrix}.
\label{eq:modal}
\end{equation}
Since $\operatorname{tr}A_\lambda=-(\mF+c\lambda)<0$ and
$\det A_\lambda=c\,\mF\lambda-\vt_1\vt_2$, $A_\lambda$ is Hurwitz \emph{iff}
$\lambda>\sigmadyn:=\vt_1\vt_2/(c\,\mF)$, and its decay rate is
\begin{equation}
\alpha(\lambda)=\tfrac12\Big(\mF+c\lambda-
\sqrt{(\mF-c\lambda)^2+4\vt_1\vt_2}\Big).
\label{eq:rate}
\end{equation}
$\alpha$ is strictly increasing ($\alpha'=\tfrac c2[1-(c\lambda-\mF)/
\sqrt{(\mF-c\lambda)^2+4\vt_1\vt_2}]>0$) with $\alpha(\sigmadyn)=0$, so the whole
interconnection is exponentially stable iff $\lamtwo>\sigmadyn$, at rate $\ge\alpha(\lamtwo)$.
\end{proposition}
Chaining this with the geometric transfer closes the loop from assignment to stability, but the
two links must be kept apart. The relaxed selector supplies a \emph{design and existence}
certificate: $\LamE\ge\sigmastar$ says the fiber contains a fractional composition meeting the
requirement. The \emph{physical} guarantee attaches only to a realizable team: for any integer
$\hat z\in\mathcal X\cap\{0,1\}^n$ that passes direct re-certification with
$\lamtwo(\Lap(\hat z))\ge\sigmastar>\sigmadyn$, Lemma~\ref{lem:bridge} applies and yields
$\lamtwo(\Lgeo(q,\hat z))\ge\sigmastar$ for \emph{every} admissible pose, hence a uniform
guaranteed rate $\alpha(\sigmastar)>0$. Lemma~\ref{lem:bridge} is stated for integer $\hat z$
and is not invoked at the fractional $z^{\star}_{\mathrm{WISE}}$. This is a \emph{reduced
information-layer} model---not robot, load or actuator dynamics---and no online distributed
realization is analyzed here.

\subsection{The price of self-sustainability}
\label{sec:price}
The lexicographic selector of Theorem~\ref{thm:existence} is exact; a scalarised
$V_\varepsilon=V+\varepsilon\lamtwo$ recovers it only as $\varepsilon\to0$
(Tikhonov~\cite{facchinei2003fvi}: $\|Bz_\varepsilon-y^{\star}\|\le
\sqrt{2\varepsilon\,\mathrm{osc}_{X_{\mathrm f}}(\lamtwo)/\alpha}$, with
$\mathrm{osc}_{X_{\mathrm f}}(\lamtwo)=\max_{X_{\mathrm f}}\lamtwo-\min_{X_{\mathrm f}}\lamtwo
=\LamX-\min_{X_{\mathrm f}}\lamtwo$), so a finite $\varepsilon$ generally moves the
aggregate---a Pareto scalarisation, not the lexicographic form.

Beyond the free fiber the requirement carries a productive cost. Define the
\emph{price of self-sustainability}
\begin{equation}
\Price(\sigma)=V^{\star}-\max\{V(z):z\in X_{\mathrm f},\ \lamtwo(\Lap(z))\ge\sigma\}
\label{eq:price}
\end{equation}
($\Price(\sigma)=+\infty$ when the set is empty), and the two spectral capacities
$\LamE=\max_{z\in\Emanifold}\lamtwo(\Lap(z))$ and
$\LamX=\max_{z\in X_{\mathrm f}}\lamtwo(\Lap(z))\ge\LamE$.

\begin{theorem}[Price and the free/costly/impossible trichotomy]\label{thm:price}
$\Price$ is a nondecreasing, proper extended-value convex function, and
\begin{equation}
\Price(\sigma)=
\begin{cases}
0, & \sigma\le\LamE\ \ (\text{free}),\\[1pt]
\text{finite}>0, & \LamE<\sigma\le\LamX\ \ (\text{costly}),\\[1pt]
+\infty, & \sigma>\LamX\ \ (\text{impossible}).
\end{cases}
\label{eq:trichotomy}
\end{equation}
At any $\sigma<\LamX$ admitting a strictly feasible point ($\exists z\in X_{\mathrm f}$ with
$\lamtwo(\Lap(z))>\sigma$, i.e.\ Slater holds), the optimal multiplier $\pi^{\star}$ of the
connectivity constraint satisfies $\pi^{\star}\in\partial\Price(\sigma)$: the marginal
productive price of one unit of connectivity, in the units of $V$.
\end{theorem}
\begin{proof}[Proof sketch]
The super-level set (Lemma~\ref{lem:convex}) shrinks in $\sigma$, so $\Price$ is nondecreasing;
$\sigma$ enters linearly, so $\Price$ is a convex perturbation value with subdifferential the
optimal multiplier set under Slater~\cite{boyd2004convex}. The cases follow from
$\sigma\le\LamE$ ($\Price=0$), $\sigma>\LamX$ (infeasible), and between.
\end{proof}

\subsection{Finite-team recovery}
A sufficient margin test preserves \emph{joint feasibility} after integer recovery (it does
not by itself preserve the aggregate). Let $z^{\star}$ be a relaxed WISE and define the
wrench and connectivity slacks and the sensitivity constants
\begin{equation}
\begin{aligned}
&m_w=\min_j\big((H_wz^{\star})_j-d_j\big),\quad
m_\lambda=\lamtwo(\Lap(z^{\star}))-\sigmastar,\\
&\kappa_w=\|H_w\|_{2\to\infty},\qquad
\kappa_L=\sup_{\|d\|=1}\|D\Lap[d]\|_2 .
\end{aligned}
\label{eq:margins}
\end{equation}
\begin{proposition}[Margin-based preservation of joint feasibility]\label{prop:integer}
Let $\hat z$ be any integer assignment meeting budgets, integrality and occupancy exactly
with $\|\hat z-z^{\star}\|_2\le r$. If
\begin{equation}
r<\min\!\Big\{\tfrac{m_w}{\kappa_w},\ \tfrac{m_\lambda}{\kappa_L}\Big\},
\label{eq:intmargin}
\end{equation}
then $\hat z\in X_{\mathrm f}$ and $\lamtwo(\Lap(\hat z))\ge\sigmastar$: $\hat z$ is an integer
self-sustaining assignment. If additionally $B\hat z=y^{\star}$ it is a zero-cost integer WISE;
otherwise it has \emph{recovery loss} $\ell(\hat z)=V^{\star}-V(\hat z)$. The optimal
integrality gap $g_{\mathbb Z}=V^{\star}-V_{\mathbb Z}^{\star}$ is a distinct quantity,
evaluated by the exact oracle (Section~\ref{sec:exp}).
\end{proposition}
\begin{proof}[Proof sketch]
Row-wise $(H_w\hat z)_j\ge d_j+m_w-\kappa_w r>d_j$ gives $\hat z\in X_{\mathrm f}$; and, as
$\Lap$ is affine, Weyl's inequality gives $\lamtwo(\Lap(\hat z))\ge\sigmastar+m_\lambda-
\kappa_L r>\sigmastar$.
\end{proof}

\begin{remark}[Rounding as a search]\label{rem:integer}
Proposition~\ref{prop:integer} is a checkable sufficient condition, not an existence
guarantee: best-of-$R$ rounding \emph{searches} for a feasible $\hat z$ and \emph{directly}
re-checks it against budgets, occupancy, the lifted wrench LP and $\lamtwo\ge\sigmastar$ (no
$O(1/N)$ claim). The radius~\eqref{eq:intmargin} is a separate \emph{worst-case} certificate,
dominated by the spectral term $m_\lambda/\kappa_L$ (with $\kappa_L$ bounded by
Cauchy--Schwarz, hence itself conservative); Section~\ref{sec:exp} reports how few recoveries
it certifies, so direct re-certification is the operative finite-team test.
\end{remark}
