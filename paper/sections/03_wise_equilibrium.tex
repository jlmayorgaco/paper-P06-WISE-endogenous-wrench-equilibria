\section{Optimal-Fiber Networkability and WISE Selection}
\label{sec:wise}

\subsection{Productive value and the optimal composition fiber}
Wrench feasibility guarantees each load its physical service; the productive value then
allocates served capacity through the aggregate $y=Bz$, $y_k=\sum_{i,h}c_{\tau(i)}z_{ikh}$
(distinct from the wrench aggregate: $y_k$ is load $k$'s reserve beyond the wrench floor):
\begin{equation}
V(z)=\phi(Bz),\qquad
\phi(y)=\sum_k\big(v_k\,y_k-\tfrac12\alpha\,y_k^2\big),
\label{eq:value}
\end{equation}
so $\phi$ is strictly concave ($\alpha>0$, $y_{\mathrm{ref}}=v/\alpha$), pinning down a
unique optimal aggregate. What $V$ leaves undetermined is the subject of this section.

Because $V$ is concave on the convex $X_{\mathrm f}$, the variational solution set coincides
with the optimal set, $\operatorname{SOL}(X_{\mathrm f},-\nabla V)=\arg\max_{X_{\mathrm f}}V
=\Emanifold$~\cite{facchinei2003fvi}: a \emph{set-valued} productive equilibrium, unique in
aggregate but not in composition, which WISE refines lexicographically.

\begin{theorem}[Composition degeneracy]\label{thm:degeneracy}
Since $\phi$ is strictly concave on the compact convex $BX_{\mathrm f}$, the optimal aggregate
$y^{\star}=Bz^{\star}$ is unique, but the optimal set
$\Emanifold=\{z\in X_{\mathrm f}:Bz=y^{\star}\}$ is a compact convex polytope whose local
dimension at a relative-interior point $\bar z$ is
\begin{equation}
\dim\Emanifold=\dim\!\big(\ker A\cap\ker B\cap\ker G_I\big)
=n-\operatorname{rank}[A;B;G_I],
\label{eq:dim}
\end{equation}
where $Az=b$ is budget conservation, $Bz=y^{\star}$ the served aggregate, $G_Iz=h_I$ the
inequalities of $X_{\mathrm f}$ active at $\bar z$, and $n=N|\mathcal A|$. Thus $\Emanifold$ is
positive-dimensional iff the \emph{productive neutral space}
$N_p(\bar z)=\ker A\cap\ker B\cap\ker G_I$ is nontrivial.
\end{theorem}
\begin{proof}[Proof sketch]
At $\bar z\in\operatorname{relint}\Emanifold$ the feasible perturbations are exactly
$N_p(\bar z)$, so $\Emanifold$ locally coincides with $\bar z+N_p(\bar z)$ and \eqref{eq:dim} is
rank--nullity (a general weighted $B$ can give $N_p=\{0\}$).
\end{proof}
On the $N{=}12$ instances of Sec.~\ref{sec:exp} ($n=72$, $\operatorname{rank}[A;B]=13$, no
active inequality) this gives $\dim\Emanifold=59$, certified to residuals $<10^{-15}$; a
boundary optimum where wrench facets bind gives a smaller fiber.

Positive fiber dimension is \emph{equivalent} to loss of strong monotonicity on the active
face: with $P_F$ of \emph{orthonormal} columns spanning $\ker A\cap\ker G_I$, the
pseudo-gradient $F(z)=\alpha B^{\top}(Bz-y_{\mathrm{ref}})$ gives
$\mu_F=\alpha\lambda_{\min}(P_F^{\top}B^{\top}BP_F)$, so $\mu_F=0\iff N_p(\bar z)\neq\{0\}$,
unlike strongly contractive aggregative games~\cite{martinezpiazuelo2022}.

Figure~\ref{fig:concept} depicts the affine geometry underlying
Thm.~\ref{thm:networkability}; Fig.~\ref{fig:hero}(a) illustrates an analogous exchange in a
\emph{separate} pair-dependent stress test and does not validate the directional certificate.
For the affine model the sign of $\Gamma_{\Emanifold}$ decides whether a
connectivity-improving neutral direction exists. Orient every inequality of $X_{\mathrm f}$ as $Gz\ge h$,
with $G_I$ the rows active at $\bar z$. The cone may be taken in the lifted variables at
\emph{any} lift: for the polyhedron $\widehat\Emanifold$ lifted by \eqref{eq:tensor},
$T_{\widehat\Emanifold}(\hat x)=\operatorname{cone}(\widehat\Emanifold-\hat x)$ and a linear
map commutes with the conic hull, so
$\prz(T_{\widehat\Emanifold}(z,\xi))=T_{\Emanifold}(z)$ \emph{exactly}; hence
$\Gamma_{\Emanifold}(\bar z)=\max\{\lambda_{\min}(U_2^{\top}D\Lap[d_z]U_2):
(d_z,d_\xi)\in T_{\widehat\Emanifold}(\bar z,\xi),\ \|d_z\|_2\le1\}$ is exact there. On the
$\Gamma_{\Emanifold}$-validation instances of Sec.~\ref{sec:exp} $G_I$ is empty, so
$T_{\Emanifold}(\bar z)=\ker A\cap\ker B$; E4 exercises a binding facet.

\begin{definition}[Free-networkability modulus]\label{def:gamma}
For $\bar z\in\Emanifold$ let $T_{\Emanifold}(\bar z)=\{d:Ad{=}0,\ Bd{=}0,\ G_Id\ge0\}$ be the
tangent cone, and with $\widetilde L(\bar z)=Q^{\top}\Lap(\bar z)Q$ (so
$\lambda_{\min}(\widetilde L)=\lamtwo(\Lap)$) let $\widetilde U_2$ span its
minimum-eigenvalue eigenspace; set $U_2(\bar z)=Q\widetilde U_2\in\R^{N\times m}$,
$m=\dim\ker(\widetilde L(\bar z)-\lamtwo I)$,
so $U_2^{\top}\mathbf1=0$ \emph{by construction}, which matters because on a disconnected
graph $\lamtwo(\Lap)=0$ is degenerate \emph{with} the consensus mode. As $\Lap$ is affine,
$D\Lap[d]=\sum_{i,r}d_{ir}L_{ir}$ and~\cite{lewis1996spectral} gives
\begin{equation}
D\lamtwo(\bar z)[d]=\lambda_{\min}\!\big(U_2^{\top}D\Lap[d]\,U_2\big),
\label{eq:dirderiv}
\end{equation}
reducing to $\langle g_\lambda,d\rangle$, $[g_\lambda]_{ir}=v^{\top}L_{ir}v$, when $m=1$ with
Fiedler vector $v=U_2$. The \emph{free-networkability modulus} is
\begin{equation}
\Gamma_{\Emanifold}(\bar z)=\max\big\{D\lamtwo(\bar z)[d]:\
d\in T_{\Emanifold}(\bar z),\ \|d\|_2\le1\big\}.
\label{eq:gamma}
\end{equation}
\end{definition}

\begin{theorem}[Optimal-fiber networkability: a local test certifies global optimality]
\label{thm:networkability}
Let $\bar z\in\Emanifold$, with $\Emanifold$ the compact convex \emph{polyhedral} fiber of
Theorem~\ref{thm:degeneracy}. Then $\Gamma_{\Emanifold}(\bar z)\ge0$ and
\begin{equation}
\Gamma_{\Emanifold}(\bar z)>0
\iff
\exists\,z'\in\Emanifold:\ \lamtwo(\Lap(z'))>\lamtwo(\Lap(\bar z)),
\label{eq:iff}
\end{equation}
equivalently $\Gamma_{\Emanifold}(\bar z)=0\iff\lamtwo(\Lap(\bar z))=\LamE$: a \emph{local}
test certifying \emph{global} connectivity optimality over the fiber, and bounding the gap by
$\LamE-\lamtwo(\Lap(\bar z))\le\Gamma_{\Emanifold}(\bar z)\|z^{\star}_{\mathrm{WISE}}-\bar
z\|_2$. At $\bar z\in\operatorname{relint}\Emanifold$ with $\lamtwo$ simple, $T_{\Emanifold}=N_p$ and
\eqref{eq:gamma} is closed-form, $\Gamma_{\Emanifold}(\bar z)=\|\Pi_{N_p}g_\lambda\|_2$; in
general it is the small convex SDP $\max\{t:d\in T_{\Emanifold},\|d\|_2\le1,
U_2^{\top}D\Lap[d]U_2\succeq tI\}$.
\end{theorem}
\begin{proof}[Proof sketch]
$\Gamma_{\Emanifold}\ge0$ as $d=0$ is admissible, and $f=\lamtwo\circ\Lap$ is finite concave,
so $Df(\bar z)[\cdot]$ is positively homogeneous and overestimates increments. ($\Leftarrow$)
An improving $z'$ gives $d_0=z'-\bar z\in T_{\Emanifold}(\bar z)$ with $Df(\bar z)[d_0]>0$.
($\Rightarrow$) polyhedrality gives $T_{\Emanifold}(\bar z)=\operatorname{cone}
(\Emanifold-\bar z)$, so $\bar z+td\in\Emanifold$ improves $f$ for small $t>0$; the gap bound
is concavity plus homogeneity.
\end{proof}

\subsection{The WISE refinement}
\begin{definition}[WISE and its integer counterparts]\label{def:wise}
A profile $z^{\star}\in\Emanifold$ is a \emph{Wrench- and Information-Self-Sustaining
Equilibrium} if $\lamtwo(\Lap(z^{\star}))\ge\sigmastar$, i.e.\
$z^{\star}\in\Essrel:=\Emanifold\cap\Cdelta$, $\Cdelta=\{z\in X_{\mathrm f}:
\lamtwo(\Lap(z))\ge\sigmastar\}$ (convex and closed by concavity). For finite teams the
self-sustaining feasible set $\mathcal F_{\mathbb Z}=\{z\in X_{\mathrm f}\cap\{0,1\}^{n}:
\lamtwo\ge\sigmastar\}$, its optimum $V_{\mathbb Z}^{\mathrm{ss}}$ and the \emph{zero-cost
integer WISE} $\Essint=\mathcal F_{\mathbb Z}\cap\{Bz=y^{\star}\}$ differ. As $\mathcal F_{\mathbb Z}$ imposes connectivity as well as integrality,
$g_{\mathbb Z}^{\mathrm{ss}}=V^{\star}-V_{\mathbb Z}^{\mathrm{ss}}$ is the \emph{integer
self-sustaining gap}, not a pure integrality gap; \emph{zero-cost} means
$\|B\hat z-y^{\star}\|_\infty\le\tau_B=10^{-8}$. Every WISE
\emph{is} a productive variational equilibrium: connectivity is a \emph{lexicographic
refinement} of $\operatorname{SOL}(X_{\mathrm f},F)=\Emanifold$, not an extra strategic
condition. Equivalently $z^{\star}_{\mathrm{WISE}}$ satisfies
$0\in\partial[-\lamtwo(\Lap(z^{\star}_{\mathrm{WISE}}))]+N_{\Emanifold}
(z^{\star}_{\mathrm{WISE}})$, whose primal test is $\Gamma_{\Emanifold}=0$
(Thm.~\ref{thm:networkability}).
\end{definition}

\begin{theorem}[Relaxed WISE existence and selection as an SDP]\label{thm:existence}
\emph{Stage 1} fixes $y^{\star}=Bz^{\star}$ and \emph{Stage 2} selects connectivity on the
fiber: $z^{\star}_{\mathrm{WISE}}\in\arg\max_{\Emanifold}\lamtwo(\Lap(z))$ solves, in the
lifted wrench variables,
\begin{equation}
\LamE=\max_{z,\,\xi,\,t}\ t\quad\text{s.t.}\quad
z\in\mathcal X,\ Bz=y^{\star},\ \eqref{eq:tensor},\ Q^{\top}\Lap(z)\,Q\succeq tI,
\label{eq:sdp}
\end{equation}
convex (affine constraints, concave $\lamtwo\!\circ\!\Lap$). A \emph{relaxed} WISE exists iff
the value clears the requirement,
\begin{equation}
\Essrel\neq\varnothing\iff\LamE\ge\sigmastar,
\label{eq:exist}
\end{equation}
and any maximizer preserves the productive optimum exactly, with $V=V^{\star}$; the refinement
is \emph{strict} when $\min_{\Emanifold}\lamtwo<\sigmastar\le\LamE$. If $\arg\max_{\Emanifold}\lamtwo$ (equivalently $\{\Gamma_{\Emanifold}=0\}$) is
non-singleton, the strongly convex least-role-change tie-break
$\min\tfrac12\|z-z_{\mathrm{prev}}\|_W^2$ fixes a \emph{unique} selector.
\end{theorem}
\begin{proof}[Proof sketch]
The constraints are convex, so \eqref{eq:sdp} is an SDP with value $\LamE$ attained on the
compact $\Emanifold$, and feasibility forces $Bz^{\star}_{\mathrm{WISE}}=y^{\star}$.
\end{proof}

\subsection{Information-layer consequence}
\begin{proposition}[Reduced information-layer interface threshold]\label{thm:stability}
In the coordinates $a,b\in\R^{N-1}$ of $\mathbf1^{\perp}$, let a mode-aligned coordination
error $a$ be driven by a consensus estimator error $b$ diffusing over
$\widetilde L(z)=Q^{\top}\Lap(z)Q$ ($a$ is not the served-aggregate error):
\begin{equation}
\dot a=-\mF\,a+\vt_1 b,\qquad
\dot b=\vt_2\,a-c\,\widetilde L(z)\,b,
\label{eq:interconnection}
\end{equation}
with damping $\mF>0$, consensus gain $c>0$ and \emph{isotropic} couplings $\vt_1I,\vt_2I$,
$\vt_1,\vt_2>0$, so \eqref{eq:interconnection} block-diagonalizes in the eigenbasis of
$\widetilde L$: each eigenvalue $\lambda$ contributes an independent pair
$[\dot a_\lambda;\dot b_\lambda]=A_\lambda[a_\lambda;b_\lambda]$ with
$A_\lambda=\big(\begin{smallmatrix}-\mF&\vt_1\\ \vt_2&-c\lambda\end{smallmatrix}\big)$.
As $\operatorname{tr}A_\lambda<0$ and $\det A_\lambda=c\mF\lambda-\vt_1\vt_2$, $A_\lambda$ is
Hurwitz \emph{iff} $\lambda>\sigmadyn:=\vt_1\vt_2/(c\mF)$, with decay rate
\begin{equation}
\alpha(\lambda)=\tfrac12\Big(\mF+c\lambda-
\sqrt{(\mF-c\lambda)^2+4\vt_1\vt_2}\Big).
\label{eq:rate}
\end{equation}
$\alpha$ is strictly increasing with $\alpha(\sigmadyn)=0$, so the interconnection is
exponentially stable iff $\lamtwo>\sigmadyn$, at rate $\ge\alpha(\lamtwo)$. The fixed gains
$(\mF,c,\vt_1,\vt_2)=(1,1,0.5,0.5)$ used throughout give $\sigmadyn=0.25$ and
$\alpha(0.30)=0.0397$.
\end{proposition}

This assumes a \emph{fixed} Laplacian; a moving team drives \eqref{eq:interconnection} with
$Q^{\top}\Lgeo(q(t))Q$, whose eigenbasis rotates. A common Lyapunov function restores the
guarantee at the same rate.

\begin{corollary}[Moving graph, unchanged certified lower-rate bound]\label{cor:tv}
Let \eqref{eq:interconnection} be driven by $\Lgeo(q(t))$ with
$Q^{\top}\Lgeo(q(t))Q\succeq\sigmastar I$ for every $t$ of the operational window. Then, with
$\mathcal V_c=\tfrac12(\vt_2\|a\|^2+\vt_1\|b\|^2)$ and $\|\cdot\|_P=\sqrt{2\mathcal V_c}$,
$\dot{\mathcal V}_c\le-2\alpha(\sigmastar)\mathcal V_c$, i.e.\
$\|[a,b](t)\|_P\le e^{-\alpha(\sigmastar)t}\|[a,b](0)\|_P$, with $\alpha$ \emph{exactly}
\eqref{eq:rate}, positive iff $\sigmastar>\sigmadyn$.
\end{corollary}
\begin{proof}[Proof sketch]
With $b^{\top}Q^{\top}\Lgeo Qb\ge\sigmastar\|b\|^2$ and $P=\operatorname{diag}(\vt_2,\vt_1)$,
$\dot{\mathcal V}_c\le-[a;b]^{\top}(M{\otimes}I)[a;b]\le-2\lambda_{\min}(M,P)\mathcal V_c$ for
$M=(\vt_2\mF,-\vt_1\vt_2;-\vt_1\vt_2,\vt_1c\sigmastar)$; as
$P^{-1/2}MP^{-1/2}=(\mF,-\sqrt{\vt_1\vt_2};-\sqrt{\vt_1\vt_2},c\sigmastar)$, that generalized
eigenvalue \emph{is} $\alpha(\sigmastar)$: $\mathcal V_c$ symmetrizes $A_\lambda$, so the
moving graph costs no rate.
\end{proof}
The relaxed and physical guarantees apply at different stages. The selector gives a
\emph{design and existence} certificate ($\LamE\ge\sigmastar$); the \emph{physical} guarantee
attaches only to a realizable team: for integer $\hat z$ passing re-certification with
$\lamtwo(\Lap(\hat z))\ge\sigmastar>\sigmadyn$, Lemma~\ref{lem:bridge} yields
$\lamtwo(\Lgeo(q,\hat z))\ge\sigmastar$ at \emph{every} admissible pose, hence the uniform rate
$\alpha(\sigmastar)>0$ of Cor.~\ref{cor:tv}, never invoked at the fractional
$z^{\star}_{\mathrm{WISE}}$. Equation~\eqref{eq:interconnection} is an assumed \emph{interface
abstraction}, not a model derived here from any revision protocol or estimator.

\subsection{The price of self-sustainability}
\label{sec:price}
A scalarized $V+\varepsilon\lamtwo$ recovers the lexicographic selector only as
$\varepsilon\to0$~\cite{facchinei2003fvi}. Beyond the free fiber the requirement carries a
cost: with $\Price(\sigma)=V^{\star}-\max\{V(z):z\in X_{\mathrm f},
\lamtwo(\Lap(z))\ge\sigma\}$ and $\LamE\le\LamX=\max_{X_{\mathrm f}}\lamtwo$, the super-level
set is convex and shrinks in $\sigma$, which enters linearly, so $\Price$ is a nondecreasing
convex perturbation value: $\Price=0$ for $\sigma\le\LamE$ (\emph{free}), finite and positive
for $\LamE<\sigma\le\LamX$ (\emph{costly}), $+\infty$ beyond (\emph{impossible}), with
$\pilam^{\star}\in\partial\Price(\sigma)$ under Slater.

The spectral reserve yields the following attenuation guarantee, a certified budget for
degradation of the relays themselves.

\begin{corollary}[Certified attenuation budget]\label{cor:budget}
Attenuate every gated relay conductance by $\rho\in[0,1]$:
$\Lap_\rho(z)=L_0+(1-\rho)\sum_{i,r}z_{ir}L_{ir}=(1-\rho)\Lap(z)+\rho L_0$. Concavity of
$\lamtwo$ gives $\lamtwo(\Lap_\rho)\ge(1-\rho)\lamtwo(\Lap(z))+\rho\lamtwo(L_0)$, so
$\lamtwo(\Lap_\rho)\ge\sigmastar$ for every
\begin{equation}
\rho\le\rho_{\mathrm{cert}}(z)=
\frac{\lamtwo(\Lap(z))-\sigmastar}{\lamtwo(\Lap(z))-\lamtwo(L_0)},
\label{eq:budget}
\end{equation}
reducing to $1-\sigmastar/\lamtwo(\Lap(z))$ when the base graph is disconnected. The ratio is
well defined when $\lamtwo(\Lap(z))>\lamtwo(L_0)$; if the two coincide, attenuation leaves the
lower bound unchanged. Nonuniform
$\rho_{ir}\le\rho$ obeys the same bound since $L_{ir}\succeq0$. For
$\sigmastar>\lamtwo(L_0)$---the regime where relays are needed at all---$\rho_{\mathrm{cert}}$
is increasing in $\lamtwo(\Lap(z))$, so \emph{WISE maximizes the certified degradation budget
over the whole productive-optimum fiber.}
\end{corollary}

\subsection{Finite-team recovery}
We distinguish three quantities: the \emph{relaxed} fiber optimum, where
Thm.~\ref{thm:existence} gives $\Delta V=0$ exactly; the \emph{attainable integer} optimum,
whose shortfall is $g_{\mathbb Z}^{\mathrm{ss}}$; and the \emph{recovery loss}
$V^{\star}-V(\hat z)$ of one rounding heuristic. So \eqref{eq:sdp} certifies existence on the
fractional fiber and sets a target for recovery, but does \emph{not} guarantee integer
attainability: Sec.~\ref{sec:exp} exhibits adversarial instances where it fails. Rounding
searches for an integer $\hat z$ and re-checks it directly against budgets, occupancy,
\eqref{eq:tensor} and $\lamtwo\ge\sigmastar$; if $B\hat z=y^{\star}$ too it is a zero-cost
integer WISE.
