\section{Self-Sustaining Equilibria}
\label{sec:wise}

\subsection{Productive value and the optimal composition fiber}
The team's productive value depends on the composition only through the
\emph{served-capacity} aggregate $y=Bz$, $y_k=\sum_{\tau,h}c_\tau z_{\tau kh}$ (the
capacity recruited to load $k$):
\begin{equation}
V(z)=\phi(Bz),\qquad
\phi(y)=\sum_k\big(v_k\,y_k-\tfrac12\alpha\,y_k^2\big),
\label{eq:value}
\end{equation}
so $\phi(y)=-\tfrac12\|y-y_{\mathrm{ref}}\|_H^2+\mathrm{const}$ with $H=\alpha
I\succ0$ and $y_{\mathrm{ref}}=v/\alpha$. It is strictly concave
($\nabla^2\phi=-\alpha I\prec0$) and so pins down a unique optimal aggregate; a
deficit penalty such as $-\|[d-s]_+\|^2$ would be flat on the feasible set and
leave the aggregate undetermined. The game is a constrained potential population
game with payoff $\partial V/\partial z_{\tau a}$, and it is identical-interest:
the novelty lies in what $V$ leaves undetermined.

Wrench feasibility is a \emph{hard} shared constraint, its matrix $H_w$ (the
support map, $s=H_w z$) kept notationally distinct from the objective Hessian $H$:
\begin{equation}
X_{\mathrm f}=\{z\in\mathcal X:\;H_w z\ge d\}.
\label{eq:Xf}
\end{equation}
The action cost $g$ enters only as a selection refinement and is \emph{excluded}
from the degeneracy result; the productive optimum is
$\arg\max_{z\in X_{\mathrm f}}\phi(Bz)$.

\begin{lemma}[Unique productive aggregate]\label{lem:uniqueagg}
If $\phi$ is strictly concave on $BX_{\mathrm f}$ ($H=\alpha I\succ0$) and
$X_{\mathrm f}$ is nonempty compact convex, the productively optimal aggregate
$y^{\star}=Bz^{\star}$ is unique. Uniqueness is claimed for the \emph{aggregate},
not the composition.
\end{lemma}
\begin{proof}
$\phi$ strictly concave on the compact convex $BX_{\mathrm f}$ attains its maximum
at a unique $y^{\star}$: two distinct optima would have a strictly better midpoint.
\end{proof}

\begin{theorem}[Composition degeneracy]\label{thm:degeneracy}
Under Lemma~\ref{lem:uniqueagg} the aggregate $y^{\star}=Bz^{\star}$ is unique, but
the optimal set $\Emanifold=\{z\in X_{\mathrm f}:Bz=y^{\star}\}$ is a nonempty compact
convex polytope whose local dimension at a relative-interior point $\bar z$ is
\begin{equation}
\dim\Emanifold=\dim\!\big(\ker A\cap\ker B\cap\ker G_I\big)
=n-\operatorname{rank}[A;B;G_I],
\label{eq:dim}
\end{equation}
where $Az=b$ is budget conservation, $Bz=y^{\star}$ the served aggregate, $G_Iz=h_I$
the inequalities of $X_{\mathrm f}$ active at $\bar z$ (nonnegativity, occupancy, active wrench),
and $n=\dim z$; equivalently $\dim\Emanifold=\dim F-\operatorname{rank}(B|_{\mathrm{lin}F})$
for the minimal face $F\supseteq\Emanifold$. Thus $\Emanifold$ is positive-dimensional
iff the \emph{productive neutral space} $N_p(\bar z)=\ker A\cap\ker B\cap\ker G_I$ is
nontrivial. The transportation value $(T-1)(K_{\mathrm{act}}-1)$ is the special case
of equal column-sum weights with only row/column sums active.
\end{theorem}
\begin{proof}
Uniqueness of $y^{\star}$ is Lemma~\ref{lem:uniqueagg}. At
$\bar z\in\operatorname{relint}\Emanifold$ the minimal face $F$ fixes exactly the
active constraints, so the feasible perturbations are
$\{d:Ad=0,\,Bd=0,\,G_Id=0\}=N_p(\bar z)$; hence $\Emanifold$ locally coincides with
$\bar z+N_p(\bar z)$ and \eqref{eq:dim} is rank--nullity. Since $V=\phi(Bz)$ depends on
$z$ only through $Bz$, every $z\in\Emanifold$ attains the same $V$; a general weighted
$B$ can give $N_p=\{0\}$ (a single composition).
\end{proof}
For the representative instance we obtain $\dim\Emanifold=4$ out of $n=72$, with the
neutral-direction residuals $\|Ad\|,\|Bd\|,\|G_Id\|$ at machine precision. Classical
theory thus fixes \emph{how much} each load receives, not \emph{which types} provide it.

\begin{proposition}[Comparative-advantage exchange]\label{prop:cycle}
A mass-preserving four-cycle $d$ ($d_{\tau a}\!=\!+1$, $d_{\tau b}\!=\!-1$,
$d_{\tau' a}\!=\!-1$, $d_{\tau' b}\!=\!+1$) is productively neutral ($Bd=0$) iff
$B_{\tau a}-B_{\tau b}=B_{\tau' a}-B_{\tau' b}$. A neutral exchange can still change
connectivity: at a simple Fiedler eigenvalue with unit vector $v$,
$D\lamtwo(\Lap(z))[d]=\sum_e(\partial a_e/\partial z)^{\!\top}\!d\,\;v^{\top}L_e v$, and
the exchange is locally connectivity-improving when this is positive. A long-range
type can thus trade a lifting role to a short-range type at no productive cost while
raising $\lamtwo$.
\end{proposition}

\subsection{The WISE refinement}
Place the endogenous Laplacian \eqref{eq:laplacian} on $\Emanifold$.

\begin{definition}[WISE]\label{def:wise}
A profile $z^{\star}\in\Emanifold$ is a \emph{Wrench- and
Information-Self-Sustaining Equilibrium} if $\lamtwo(\Lap(z^{\star}))\ge\sigmastar$,
i.e.\ $z^{\star}\in\Ess:=\Emanifold\cap\Cdelta$, where
$\Cdelta=\{z\in X_{\mathrm f}:\lamtwo(\Lap(z))\ge\sigmastar\}$. Since
$\sigmastar=\sigmadyn+\delta>\sigmadyn$, a WISE clears the stability threshold of
Theorem~\ref{thm:stability} with margin $\delta$ and, by Lemma~\ref{lem:bridge},
keeps the \emph{geometric} graph above $\sigmadyn$. $\Cdelta$ is convex and closed
(super-level set of the concave $\lamtwo\!\circ\!\Lap$, Lemma~\ref{lem:convex}).
\end{definition}

\begin{theorem}[Existence by selection]\label{thm:existence}
Let $\LamE=\max_{z\in\Emanifold}\lamtwo(\Lap(z))$. Then
\begin{equation}
\Ess\neq\varnothing\iff\LamE\ge\sigmastar,
\label{eq:exist}
\end{equation}
and $\Ess$ is convex. When $\dim\Emanifold>0$ and
$\min_{\Emanifold}\lamtwo<\sigmastar\le\LamE$, the refinement is \emph{strict}:
some productively optimal compositions self-sustain and others do not.
\end{theorem}
\begin{proof}
$\lamtwo\!\circ\!\Lap$ is concave (Lemma~\ref{lem:convex}) and continuous on the
compact convex $\Emanifold$, so it attains its maximum $\LamE$; $\Ess$ is
nonempty iff $\LamE\ge\sigmastar$. $\Ess$ is the intersection of the polytope
$\Emanifold$ with a convex super-level set, hence convex. Strictness is immediate
when the threshold falls strictly inside the range of $\lamtwo$ over $\Emanifold$.
\end{proof}

\subsection{The self-defeat threshold, derived}
The distributed dynamics run a strategy-revision loop and a consensus estimator of
the aggregates over $\Lap(z)$. Linearising about $z^{\star}$, with $m_F$ the
strong-monotonicity constant of the revision protocol, $c$ the consensus gain, and
$\vt$ the cross-coupling by which stale aggregate estimates perturb revision, the
coupled error $e=(e_x,e_\eta)$ obeys
\begin{equation}
\dot e=J\,e,\qquad
J=\begin{bmatrix}-m_F & \vt\\[1mm] \vt & -c\,\lamtwo(\Lap(z^{\star}))\end{bmatrix}.
\label{eq:jac}
\end{equation}

\begin{theorem}[Local self-computation certificate]\label{thm:stability}
The canonical block \eqref{eq:jac} is Hurwitz iff
\begin{equation}
c\,m_F\,\lamtwo(\Lap(z^{\star}))>\vt^{2},
\quad\text{i.e.}\quad
\lamtwo>\sigmadyn:=\frac{\vt^{2}}{c\,m_F}.
\label{eq:threshold}
\end{equation}
At equality $\det J=0$ (a zero eigenvalue---\emph{not} exponential stability); below
it $J$ has a positive real eigenvalue.
\end{theorem}
\begin{proof}
$\operatorname{tr}J=-(m_F+c\lamtwo)<0$ always and $\det J=c\,m_F\,\lamtwo-\vt^{2}$;
a real $2\times2$ matrix is Hurwitz iff $\operatorname{tr}J<0$ and $\det J>0$, giving
\eqref{eq:threshold}.
\end{proof}
This is the worst-mode certificate: the vectored strategy--estimator error
diagonalises over $Q^{\top}\Lap(z^{\star})Q$ and the tightest block is the Fiedler
mode $\lamtwo$; we state \eqref{eq:threshold} for that canonical aligned block and
do \emph{not} claim ``iff'' for the full nonlinear system. WISE uses the strict
margin $\sigmastar=\sigmadyn+\delta$, so a WISE is exponentially stable, not
marginal.

\subsection{Selection, prices, and the cost of endogeneity}
\begin{proposition}[Infinitesimal-reward selection]\label{prop:selection}
For $V_\varepsilon(z)=V(z)+\varepsilon\,\lamtwo(\Lap(z))$, $\varepsilon>0$, every
maximiser $z_\varepsilon$ over $X_{\mathrm f}$ satisfies $Bz_\varepsilon\to y^{\star}$
and, along a subsequence, $z_\varepsilon\to\arg\max_{z\in\Emanifold}\lamtwo(\Lap(z))$
as $\varepsilon\downarrow0$.
\end{proposition}
\begin{proof}
$V$ is upper bounded on the compact $X_{\mathrm f}$ with optimal set $\Emanifold$
(Theorem~\ref{thm:degeneracy}); $\lamtwo\!\circ\!\Lap$ is continuous. By the
Tikhonov/$\Gamma$-selection principle, maximisers of $V+\varepsilon\lamtwo$ converge
as $\varepsilon\downarrow0$ to maximisers of $\lamtwo$ over $\arg\max V=\Emanifold$.
\end{proof}
An infinitesimal information value breaks the composition degeneracy toward the
most-connected profile, with productive loss \emph{vanishing} as
$\varepsilon\downarrow0$: quantitatively
$0\le V^{\star}-V(z_\varepsilon)\le\varepsilon\,\operatorname{osc}(\lamtwo)$ and
$\|Bz_\varepsilon-y^{\star}\|\le\sqrt{2\varepsilon\,\operatorname{osc}(\lamtwo)/\alpha}$.
Only the exact Stage-2 problem $\max_{z\in\Emanifold}\lamtwo(\Lap(z))$ leaves the
productive optimum unmoved; a finite $\varepsilon$ does not.

\begin{theorem}[Free / costly / impossible]\label{thm:trichotomy}
With $\LamE=\max_{z\in\Emanifold}\lamtwo(\Lap(z))$ and
$\LamX=\max_{z\in X_{\mathrm f}}\lamtwo(\Lap(z))$ (both attained), exactly one regime
holds: \emph{(free)} $\LamE\ge\sigmastar$: an optimal composition is
information-safe, $\Ess\neq\varnothing$, productive loss $\Ddelta=0$;
\emph{(costly)} $\LamE<\sigmastar\le\LamX$: no optimal composition is safe but a
safe feasible one exists, with $\Ddelta=V^{\star}-\max_{z\in X_{\mathrm f}\cap\Cdelta}V(z)>0$;
\emph{(impossible)} $\LamX<\sigmastar$: $X_{\mathrm f}\cap\Cdelta=\varnothing$ and
$\Ddelta=+\infty$.
\end{theorem}
\begin{proof}
Both maxima exist by Weierstrass; the cases partition the line by $\sigmastar$
relative to $\LamE\le\LamX$. In the costly case $X_{\mathrm f}\cap\Cdelta$ is nonempty
compact convex, so its maximum is attained, and since no point of $\Emanifold$ lies
in $\Cdelta$ it is strictly below $V^{\star}$.
\end{proof}
A single additive gap would be uninformative here: because $V$ is constant on the
optimal fiber $\Emanifold$, the quantity $V^{\star}-\max_{\Ess}V$ vanishes whenever a
WISE exists. The trichotomy instead measures productive loss against the feasible
margin set $X_{\mathrm f}\cap\Cdelta$, and role prices follow from the
productive-nullspace directional derivative of Proposition~\ref{prop:cycle}.

\begin{remark}[Finite-population recovery]\label{rem:integer}
The analysis is a fluid relaxation. Integer assignments (one action per robot) are
obtained by rounding and re-certified numerically; we make no $O(1/N)$ claim. Solving
with the robustness margins $s\ge d+m_W$ and $\lamtwo\ge\sigmastar=\sigmadyn+\delta$
leaves slack, so on the $N=12$ demonstration the fluid certification rate carries over
to integer robots---$100\%$ under randomised rounding---whereas naive rounding of a
zero-margin optimum does not.
\end{remark}
