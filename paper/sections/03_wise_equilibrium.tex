\section{Self-Sustaining Equilibria}
\label{sec:wise}

\subsection{Productive value and the optimal composition fiber}
The team's productive value depends on the composition only through the
\emph{served-capacity} aggregate $y=Bz$, $y_k=\sum_{\tau,h}c_\tau z_{\tau kh}$ (the
capacity recruited to load $k$):
\begin{equation}
V(z)=\phi(Bz),\qquad
\phi(y)=\sum_k\big(v_k\,y_k-\tfrac12\alpha\,y_k^2\big),
\label{eq:value}
\end{equation}
so $\phi(y)=-\tfrac12\|y-y_{\mathrm{ref}}\|_H^2+\mathrm{const}$ with $H=\alpha
I\succ0$ and $y_{\mathrm{ref}}=v/\alpha$. It is strictly concave
($\nabla^2\phi=-\alpha I\prec0$) and so pins down a unique optimal aggregate; a
deficit penalty such as $-\|[d-s]_+\|^2$ would be flat on the feasible set and
leave the aggregate undetermined. The game is a constrained potential population
game with payoff $\partial V/\partial z_{\tau a}$, and it is identical-interest:
the novelty lies in what $V$ leaves undetermined.

Wrench feasibility is a \emph{hard} shared constraint, its projected matrix $H_w$
(the wrench facets of Lemma~\ref{prop:membership}, $H_w z\ge d$) kept notationally
distinct from the objective Hessian $H$:
\begin{equation}
X_{\mathrm f}=\{z\in\mathcal X:\;H_w z\ge d\}.
\label{eq:Xf}
\end{equation}
The action cost $g$ enters only as a selection refinement and is \emph{excluded}
from the degeneracy result.

\begin{definition}[Productive variational equilibrium]\label{def:ve}
With pseudo-gradient $F(z)=-\nabla V(z)=-B^{\top}\nabla\phi(Bz)$, a profile
$z^{\star}\in X_{\mathrm f}$ is a \emph{productive variational equilibrium} if
\begin{equation}
\langle F(z^{\star}),\,z-z^{\star}\rangle\ge0\quad\forall z\in X_{\mathrm f},
\label{eq:vi}
\end{equation}
i.e.\ $z^{\star}$ solves $\mathrm{VI}(X_{\mathrm f},F)$. Since $V=\phi(Bz)$ is concave
on the convex $X_{\mathrm f}$, \eqref{eq:vi} holds iff
$z^{\star}\in\arg\max_{z\in X_{\mathrm f}}V(z)$: the productive optimum \emph{is} the
variational equilibrium~\cite{facchinei2003fvi}, tying it to the generalized-Nash
machinery.
\end{definition}

\begin{lemma}[Unique productive aggregate]\label{lem:uniqueagg}
If $\phi$ is strictly concave on $BX_{\mathrm f}$ ($H=\alpha I\succ0$) and
$X_{\mathrm f}$ is nonempty compact convex, the productively optimal aggregate
$y^{\star}=Bz^{\star}$ is unique. Uniqueness is claimed for the \emph{aggregate},
not the composition.
\end{lemma}
\begin{proof}
$\phi$ strictly concave on the compact convex $BX_{\mathrm f}$ attains its maximum
at a unique $y^{\star}$: two distinct optima would have a strictly better midpoint.
\end{proof}

\begin{theorem}[Composition degeneracy]\label{thm:degeneracy}
Under Lemma~\ref{lem:uniqueagg} the aggregate $y^{\star}=Bz^{\star}$ is unique, but
the optimal set $\Emanifold=\{z\in X_{\mathrm f}:Bz=y^{\star}\}$ is a nonempty compact
convex polytope whose local dimension at a relative-interior point $\bar z$ is
\begin{equation}
\dim\Emanifold=\dim\!\big(\ker A\cap\ker B\cap\ker G_I\big)
=n-\operatorname{rank}[A;B;G_I],
\label{eq:dim}
\end{equation}
where $Az=b$ is budget conservation, $Bz=y^{\star}$ the served aggregate, $G_Iz=h_I$
the inequalities of $X_{\mathrm f}$ active at $\bar z$ (nonnegativity, occupancy, active wrench),
and $n=\dim z$; equivalently $\dim\Emanifold=\dim F-\operatorname{rank}(B|_{\mathrm{lin}F})$
for the minimal face $F\supseteq\Emanifold$. Thus $\Emanifold$ is positive-dimensional
iff the \emph{productive neutral space} $N_p(\bar z)=\ker A\cap\ker B\cap\ker G_I$ is
nontrivial. The transportation value $(T-1)(K_{\mathrm{act}}-1)$ is the special case
of equal column-sum weights with only row/column sums active.
\end{theorem}
\begin{proof}
Uniqueness of $y^{\star}$ is Lemma~\ref{lem:uniqueagg}. At
$\bar z\in\operatorname{relint}\Emanifold$ the minimal face $F$ fixes exactly the
active constraints, so the feasible perturbations are
$\{d:Ad=0,\,Bd=0,\,G_Id=0\}=N_p(\bar z)$; hence $\Emanifold$ locally coincides with
$\bar z+N_p(\bar z)$ and \eqref{eq:dim} is rank--nullity. Since $V=\phi(Bz)$ depends on
$z$ only through $Bz$, every $z\in\Emanifold$ attains the same $V$; a general weighted
$B$ can give $N_p=\{0\}$ (a single composition).
\end{proof}
For the representative instance we obtain $\dim\Emanifold=4$ out of $n=72$, with the
neutral-direction residuals $\|Ad\|,\|Bd\|,\|G_Id\|$ at machine precision. Classical
theory thus fixes \emph{how much} each load receives, not \emph{which types} provide it.

\begin{proposition}[Comparative-advantage exchange]\label{prop:cycle}
A count-preserving four-cycle $d$ ($d_{\tau a}\!=\!+1$, $d_{\tau b}\!=\!-1$,
$d_{\tau' a}\!=\!-1$, $d_{\tau' b}\!=\!+1$) is productively neutral ($Bd=0$) iff
$B_{\tau a}-B_{\tau b}=B_{\tau' a}-B_{\tau' b}$. A neutral exchange can still change
connectivity: at a simple Fiedler eigenvalue with unit vector $v$,
$D\lamtwo(\Lap(z))[d]=\sum_{\tau,r}\gamma_{\tau r}\,d_{\tau r}\,v^{\top}L_r v$, and the
exchange is locally connectivity-improving when this is positive. A long-range type
can thus trade a lifting role to a short-range type at no productive cost while raising
$\lamtwo$.
\end{proposition}

\begin{example}[A two-type fiber]\label{ex:fiber}
Take $T=2$ types, one load of demand $y^{\star}$ served through a single slot, and one
relay site; both types can lift, but only the long-range type $\ell$ relays
effectively ($\gamma_{\ell r}>\gamma_{s r}\ge0$). Let $z(\alpha)$ place $\alpha$
long-range robots on relay while a short-range robot takes the vacated lifting role, a
productively neutral exchange (Prop.~\ref{prop:cycle}), so $Bz(\alpha)=y^{\star}$ and
$V(z(\alpha))\equiv V^{\star}$ for all admissible $\alpha$. Along this arc
$\lamtwo(\Lap(z(\alpha)))$ is concave and increasing, so the SDP~\eqref{eq:sdp} selects
the largest feasible $\alpha$ and a WISE exists iff that value clears $\sigmastar$
(Fig.~\ref{fig:scene}, block~2).
\end{example}

\subsection{The WISE refinement}
Place the endogenous Laplacian \eqref{eq:laplacian} on $\Emanifold$.

\begin{definition}[WISE]\label{def:wise}
A profile $z^{\star}\in\Emanifold$ is a \emph{Wrench- and
Information-Self-Sustaining Equilibrium} if $\lamtwo(\Lap(z^{\star}))\ge\sigmastar$,
i.e.\ $z^{\star}\in\Ess:=\Emanifold\cap\Cdelta$, where
$\Cdelta=\{z\in X_{\mathrm f}:\lamtwo(\Lap(z))\ge\sigmastar\}$. Since
$\sigmastar=\sigmadyn+\delta>\sigmadyn$, a WISE clears the stability threshold of
Theorem~\ref{thm:stability} with margin $\delta$ and, by Lemma~\ref{lem:bridge},
keeps the \emph{geometric} graph above $\sigmadyn$. $\Cdelta$ is convex and closed
(super-level set of the concave $\lamtwo\!\circ\!\Lap$, Lemma~\ref{lem:convex}).
\end{definition}

\begin{theorem}[WISE existence and selection as an SDP]\label{thm:existence}
Fix the unique optimal aggregate $y^{\star}$ (Lemma~\ref{lem:uniqueagg}). The
lexicographic selector $z^{\star}_{\mathrm{WISE}}\in
\arg\max_{z\in\Emanifold}\lamtwo(\Lap(z))$ is the solution of the semidefinite program
\begin{equation}
\LamE=\max_{z,\;t}\ t\quad\text{s.t.}\quad
z\in X_{\mathrm f},\ \ Bz=y^{\star},\ \ Q^{\top}\Lap(z)\,Q\succeq tI,
\label{eq:sdp}
\end{equation}
convex by Lemma~\ref{lem:convex} (affine constraints, concave $\lamtwo\!\circ\!\Lap$).
A WISE exists iff the optimal value clears the requirement,
\begin{equation}
\Ess\neq\varnothing\iff\LamE\ge\sigmastar,
\label{eq:exist}
\end{equation}
and then $z^{\star}_{\mathrm{WISE}}$ is a WISE that preserves the productive optimum
exactly, $Bz^{\star}_{\mathrm{WISE}}=y^{\star}$. When
$\min_{\Emanifold}\lamtwo<\sigmastar\le\LamE$ the refinement is \emph{strict}: some
optimal compositions self-sustain and others do not.
\end{theorem}
\begin{proof}
$X_{\mathrm f}$ and $Bz=y^{\star}$ are affine and, by Lemma~\ref{lem:convex},
$\{(z,t):Q^{\top}\Lap(z)Q\succeq tI\}$ is convex, so \eqref{eq:sdp} is an SDP; its value
$\LamE=\max_{z\in\Emanifold}\lamtwo(\Lap(z))$ is attained on the compact $\Emanifold$
(Weierstrass). $\Ess=\Emanifold\cap\Cdelta$ is nonempty iff this maximum reaches
$\sigmastar$, giving \eqref{eq:exist}; feasibility forces
$Bz^{\star}_{\mathrm{WISE}}=y^{\star}$. Strictness holds when $\sigmastar$ lies
strictly inside the range of $\lamtwo$ over $\Emanifold$.
\end{proof}

\subsection{Aggregate--estimator stability, derived}
The decentralized realization~\eqref{eq:pd} couples a served-aggregate revision error
$\xi$ with a consensus-estimator error $\eta$. Linearising~\eqref{eq:pd} about
$z^{\star}$ yields
\begin{equation}
\begin{bmatrix}\dot\xi\\ \dot\eta\end{bmatrix}
=\mathcal J\begin{bmatrix}\xi\\ \eta\end{bmatrix},\qquad
\mathcal J=\begin{bmatrix}-M & \Theta\\[1mm] \Theta^{\top} &
-c\,(\bar L_\perp\!\otimes\! I_p)\end{bmatrix},
\label{eq:jac}
\end{equation}
with $\bar L_\perp=Q^{\top}\Lap(z^{\star})Q$, $M\succeq\mF I$ the curvature of $V$ in the
served aggregate along $X_{\mathrm f}$, $\|\Theta\|_2\le\vt$ the cross-coupling, and
$c$ the consensus gain.

\begin{theorem}[Aggregate--estimator stability]\label{thm:stability}
$\mathcal J$ is symmetric; by the Schur complement of its $-M\prec0$ block it is Hurwitz
iff $c(\bar L_\perp\!\otimes\! I_p)\succ\Theta^{\top}M^{-1}\Theta$. In particular the
spectral condition
\begin{equation}
c\,\mF\,\lamtwo(\Lap(z^{\star}))>\vt^{2},
\quad\text{i.e.}\quad
\lamtwo>\sigmadyn:=\frac{\vt^{2}}{c\,\mF},
\label{eq:threshold}
\end{equation}
is \emph{sufficient} for exponential stability of the subsystem, and is also
\emph{necessary} in the isotropic modal case $M=\mF I$, $\Theta=\vt I$, where it reduces
to the Fiedler mode.
\end{theorem}
\begin{proof}
Symmetry is structural. With $-M\prec0$ invertible, $\mathcal J\prec0$ iff the Schur
complement $-c(\bar L_\perp\!\otimes\! I_p)+\Theta^{\top}M^{-1}\Theta\prec0$. Since
$\bar L_\perp\succeq\lamtwo I$ and
$\Theta^{\top}M^{-1}\Theta\preceq(\vt^2/\mF)I$, the bound $c\,\mF\,\lamtwo>\vt^2$
implies it; at $M=\mF I,\Theta=\vt I$ the two bounds coincide on the Fiedler mode, making
it necessary.
\end{proof}
The claim is about the served-aggregate/estimator subsystem, \emph{not} exponential
stability of $z^{\star}$ itself: on the positive-dimensional fiber $\Emanifold$ the
neutral composition directions stay marginal unless regularised. Because WISE uses the
strict margin $\sigmastar=\sigmadyn+\delta$, the subsystem clears $\sigmadyn$ with room.

\subsection{Selection, prices, and the cost of endogeneity}
The lexicographic selector of Theorem~\ref{thm:existence} is exact. A scalarised
alternative $V_\varepsilon(z)=V(z)+\varepsilon\lamtwo(\Lap(z))$ recovers it only in the
limit: any maximiser $z_\varepsilon$ obeys
$0\le V^{\star}-V(z_\varepsilon)\le\varepsilon\operatorname{osc}(\lamtwo)$ and
$\|Bz_\varepsilon-y^{\star}\|\le\sqrt{2\varepsilon\operatorname{osc}(\lamtwo)/\alpha}$
by Tikhonov selection~\cite{facchinei2003fvi}, so a finite $\varepsilon$ perturbs the
aggregate whereas the lexicographic form does not.

\emph{Prices.} The Lagrangian of \eqref{eq:sdp} carries a wrench dual $\mu\ge0$, an
aggregate dual $\nu$ ($Bz=y^{\star}$), budget/occupancy duals $\rho$, and a PSD spectral
dual $Z\succeq0$; stationarity reads
$H_w^{\top}\mu-B^{\top}\nu-A^{\top}\rho+\mathcal G^{*}(Z)=0$ with $\mathcal G^{*}$ the
adjoint of $z\mapsto Q^{\top}\Lap(z)Q$. Thus $\mu$ prices wrench feasibility and $Z$
certifies connectivity---no scalar multiplier is conflated with a marginal utility.

\begin{theorem}[Free / costly / impossible]\label{thm:trichotomy}
With $\LamE$ from \eqref{eq:sdp} and $\LamX=\max_{z\in X_{\mathrm f}}\lamtwo(\Lap(z))$
(both attained), the endogeneity cost
$\Ddelta=V^{\star}-\max_{z\in X_{\mathrm f}\cap\Cdelta}V(z)$ falls in exactly one regime:
\emph{(free)} $\LamE\ge\sigmastar$: an optimal composition is safe, $\Ess\neq\varnothing$,
$\Ddelta=0$; \emph{(costly)} $\LamE<\sigmastar\le\LamX$: no optimal composition is safe
but a safe feasible one exists, $\Ddelta>0$; \emph{(impossible)}
$\LamX<\sigmastar$: $X_{\mathrm f}\cap\Cdelta=\varnothing$, $\Ddelta=+\infty$.
\end{theorem}
\begin{proof}
Both maxima exist by Weierstrass and the cases partition the line by $\sigmastar$
relative to $\LamE\le\LamX$. In the costly case $X_{\mathrm f}\cap\Cdelta$ is nonempty
compact convex, so its maximum is attained and, as no point of $\Emanifold$ lies in
$\Cdelta$, strictly below $V^{\star}$. Measuring the gap against
$X_{\mathrm f}\cap\Cdelta$ rather than $\Ess$ (where $V\equiv V^{\star}$) is what makes
$\Ddelta$ informative.
\end{proof}

\begin{remark}[Finite-population recovery]\label{rem:integer}
The analysis is a fluid relaxation; integer assignments are recovered by rounding and
re-certified numerically (no $O(1/N)$ claim). The robustness margins $s\ge d+m_W$ and
$\lamtwo\ge\sigmastar$ leave slack, so on the $N=12$ demonstration the fluid
certification rate carries over to integer robots ($100\%$ under randomised rounding),
whereas naive rounding of a zero-margin optimum does not.
\end{remark}
